async def background_dispatcher():
    logger.info("🤖 Motor de Disparos Automáticos iniciado (Fuso: Cuiabá -04:00).")

    # Ao iniciar o serviço:
    # - Se já houver um horário agendado no futuro, mantém ele.
    # - Se o horário salvo já passou (ou estiver vazio), agenda para daqui a 1 minuto!
    now_init = datetime.now(cuiaba_tz)
    saved_next = get_config("next_dispatch_at")

    need_one_minute_grace = True
    if saved_next:
        try:
            saved_dt = datetime.strptime(saved_next, "%Y-%m-%d %H:%M:%S").replace(
                tzinfo=cuiaba_tz
            )
            if saved_dt > now_init:
                need_one_minute_grace = False
                rem_sec = int((saved_dt - now_init).total_seconds())
                logger.info(
                    f"⏳ Retomando agendamento futuro existente: próximo disparo às {saved_next} (faltam {rem_sec // 60}m {rem_sec % 60}s)."
                )
        except Exception:
            pass

    if need_one_minute_grace:
        next_run_init = (now_init + timedelta(minutes=1)).strftime("%Y-%m-%d %H:%M:%S")
        update_config("next_dispatch_at", next_run_init)
        logger.info(
            f"⏳ Horário anterior vencido ou vazio. Primeiro disparo agendado para daqui a 1 minuto ({next_run_init})."
        )

    while True:
        try:
            is_active = get_config("is_active")
            if is_active != "1":
                await asyncio.sleep(10)
                continue

            start_str = get_config("start_time") or "08:00"
            end_str = get_config("end_time") or "22:00"
            interval_min = int(get_config("interval_minutes") or "30")

            now_dt = datetime.now(cuiaba_tz)
            now_time = now_dt.time()
            start_time = time.fromisoformat(start_str)
            end_time = time.fromisoformat(end_str)

            # 1. Verifica se está dentro da janela de horário do dia
            if not (start_time <= now_time <= end_time):
                await asyncio.sleep(30)
                continue

            # 2. Verifica se já chegou o horário do próximo disparo (next_dispatch_at)
            next_dispatch_str = get_config("next_dispatch_at")
            if next_dispatch_str:
                try:
                    next_dt = datetime.strptime(
                        next_dispatch_str, "%Y-%m-%d %H:%M:%S"
                    ).replace(tzinfo=cuiaba_tz)
                    diff_seconds = (next_dt - now_dt).total_seconds()
                    if diff_seconds > 0:
                        await asyncio.sleep(min(5, diff_seconds))
                        continue
                except Exception:
                    pass

            # 3. Chegou a hora! Busca a próxima oferta pendente de hoje
            today_str = now_dt.strftime("%Y-%m-%d")
            offer = get_next_pending_offer(today_str)

            # OPÇÃO A: Se a fila de hoje estiver vazia e o auto-abastecimento estiver ativo, busca novas ofertas!
            if not offer and (get_config("auto_refill_enabled") or "1") == "1":
                refill_count = int(get_config("auto_refill_count") or "15")
                keywords_csv = get_config("auto_keywords") or ""
                logger.info(
                    f"🔄 [AUTO-ABASTECIMENTO] Fila de hoje ({today_str}) vazia! Gerando {refill_count} ofertas (Mix Shopee + ML)..."
                )

                import ml_auto

                res_refill = await asyncio.to_thread(
                    ml_auto.generate_mixed_auto_offers_to_queue,
                    refill_count,
                    today_str,
                    keywords_csv,
                )

                if res_refill["total_added"] > 0:
                    logger.info(
                        f"✅ [AUTO-ABASTECIMENTO] {res_refill['total_added']} ofertas adicionadas ({res_refill['shopee_added']} Shopee + {res_refill['ml_added']} ML)!"
                    )
                    offer = get_next_pending_offer(today_str)
                else:
                    logger.warning(
                        "⚠️ [AUTO-ABASTECIMENTO] Nenhuma oferta inédita gerada no ciclo atual."
                    )
            if offer:
                (
                    offer_id,
                    store,
                    product_url,
                    formatted_message,
                    media_url,
                    priority,
                    media_type,
                ) = offer

                is_video = (media_type == "video") or (
                    media_url and ".mp4" in media_url.lower()
                )
                logger.info(
                    f"📤 [DISPARO AUTOMÁTICO] Processando oferta ID {offer_id} ({store} | Mídia: {'VÍDEO' if is_video else 'IMAGEM'})..."
                )

                # Só sobrescreve a copy se for o template padrão da Shopee;
                # se for um texto personalizado do Telegram (com múltiplos links ou cupom), preserva o texto!
                is_standard_template = "💡 <b>POR QUE VALE A PENA?</b>" in (
                    formatted_message or ""
                )
                if priority == 1 and store.lower() == "shopee" and is_standard_template:
                    fresh_msg, fresh_image = refresh_offer_data(product_url)
                    fresh_media = media_url if (is_video and media_url) else fresh_image
                else:
                    fresh_msg, fresh_media = formatted_message, media_url

                final_media_type = "video" if is_video else "image"

                success_p = await send_whatsapp_offer(
                    WHATSAPP_GROUP_PRINCIPAL,
                    fresh_msg,
                    fresh_media,
                    final_media_type,
                )
                await asyncio.sleep(2)
                success_par = await send_whatsapp_offer(
                    WHATSAPP_GROUP_PARCERIA,
                    fresh_msg,
                    fresh_media,
                    final_media_type,
                )

                if success_p or success_par:
                    mark_offer_as_sent(offer_id)
                    logger.info(
                        f"✅ Oferta ID {offer_id} enviada e marcada como 'sent'!"
                    )
                else:
                    logger.error(f"❌ Falha ao enviar oferta ID {offer_id}.")

                # Após o envio, agenda o próximo para daqui a 'interval_min' minutos (ex: 30 min)
                next_run = (
                    datetime.now(cuiaba_tz) + timedelta(minutes=interval_min)
                ).strftime("%Y-%m-%d %H:%M:%S")
                update_config("next_dispatch_at", next_run)
                logger.info(
                    f"⏳ Aguardando {interval_min} minutos para o próximo disparo (Próximo: {next_run})..."
                )
            else:
                # Se a fila de hoje estiver vazia, limpa o timer e aguarda novas ofertas
                if next_dispatch_str != "":
                    update_config("next_dispatch_at", "")
                await asyncio.sleep(30)

        except asyncio.CancelledError:
            break
        except Exception as e:
            logger.error(f"❌ Erro no loop do dispatcher: {e}")
            await asyncio.sleep(10)


@asynccontextmanager
async def lifespan(fastapi_app: FastAPI):
    global telegram_app

    dispatcher_task = asyncio.create_task(background_dispatcher())

    if TOKEN:
        telegram_app = Application.builder().token(TOKEN).build()

        telegram_app.add_handler(
            CommandHandler(["gerar", "auto"], handle_gerar_command)
        )
        telegram_app.add_handler(
            MessageHandler(
                (filters.TEXT | filters.CAPTION) & ~filters.COMMAND, handle_message
            )
        )
        telegram_app.add_handler(
            CallbackQueryHandler(handle_callback, pattern="^type_")
        )
        telegram_app.add_handler(
            CallbackQueryHandler(handle_whatsapp_dispatch, pattern="^send_group_")
        )
        telegram_app.add_error_handler(error_handler)

        await telegram_app.initialize()
        await on_startup(telegram_app)
        await telegram_app.bot.delete_webhook(drop_pending_updates=True)
        await telegram_app.start()
        await telegram_app.updater.start_polling(drop_pending_updates=True)
        logger.info("🤖 Bot do Telegram (Shopee + ML + /gerar) iniciado com sucesso!")

    yield

    dispatcher_task.cancel()
    if telegram_app:
        await telegram_app.updater.stop()
        await telegram_app.stop()
        await on_shutdown(telegram_app)
        await telegram_app.shutdown()


app.router.lifespan_context = lifespan

if __name__ == "__main__":
    uvicorn.run(app, host="127.0.0.1", port=8000, reload=False)

@dataclass
class MLState:
    playwright: Playwright | None = None
    browser: Browser | None = None
    cookies_dict: dict | None = None
    cookies_expiration: datetime | None = None
    queue: asyncio.Queue = field(default_factory=asyncio.Queue)
    worker_task: asyncio.Task | None = None
    last_warning_date: date | None = None
    lock: asyncio.Lock = field(default_factory=asyncio.Lock)


ml_state = MLState()


def get_cookie_expiration(cookies_dict: dict) -> datetime | None:
    expirations = [
        c["expires"]
        for c in cookies_dict.get("cookies", [])
        if c.get("expires") and c["expires"] > 0
    ]
    if not expirations:
        return None
    return datetime.fromtimestamp(min(expirations))


def load_ml_cookies() -> None:
    cookies_json_str = os.environ.get("ML_COOKIES_JSON")
    if not cookies_json_str:
        logger.warning(
            "⚠️ ML_COOKIES_JSON não configurado — usando ML_AFFILIATE_COOKIE do .env."
        )
        return

    try:
        ml_state.cookies_dict = json.loads(cookies_json_str)
    except json.JSONDecodeError as e:
        logger.error(f"❌ ML_COOKIES_JSON inválido (não é um JSON válido): {e}")
        return

    ml_state.cookies_expiration = get_cookie_expiration(ml_state.cookies_dict)
    if ml_state.cookies_expiration:
        logger.info(
            f"🍪 Cookie do Mercado Livre carregado, válido até {ml_state.cookies_expiration:%d/%m/%Y %H:%M}"
        )
    else:
        logger.info(
            "🍪 Cookie do Mercado Livre carregado (sem data de expiração detectável)."
        )


async def maybe_warn_cookie_expiration(update: Update) -> None:
    if not ml_state.cookies_expiration:
        return

    today = date.today()
    if ml_state.last_warning_date == today:
        return

    days_left = (ml_state.cookies_expiration.date() - today).days
    if days_left > 3:
        return

    ml_state.last_warning_date = today

    if days_left >= 0:
        aviso = (
            f"⚠️ <b>Atenção:</b> o cookie de sessão do Mercado Livre vence em "
            f"<b>{ml_state.cookies_expiration:%d/%m/%Y}</b> (em {days_left} dia(s)). "
            f"Providencie a renovação em breve."
        )
    else:
        aviso = (
            f"🚨 <b>O cookie de sessão do Mercado Livre já venceu</b> "
            f"(vencimento em {ml_state.cookies_expiration:%d/%m/%Y}). "
            f"As conversões de link do Mercado Livre podem falhar até a renovação."
        )

    await update.message.reply_text(aviso, parse_mode="HTML")


# ---------- FUNÇÕES DE LIMPEZA E TRATAMENTO DE TEXTO ----------
def clean_corrupted_text(text: str) -> str:
    if not text:
        return ""
    cleaned = text.replace("\ufffd", "%")
    cleaned = re.sub(r"%+", "%", cleaned)
    return cleaned


def escape_html(text: str) -> str:
    if not text:
        return ""
    return text.replace("&", "&amp;").replace("<", "&lt;").replace(">", "&gt;")


def remove_affiliate_disclaimers(text: str) -> str:
    if not text:
        return ""

    patterns = [
        r"\*?\s*comprando (?:através|por meio) do link você (?:estará|está) ajudando[^\n]*",
        r"\*?\s*você me paga um café[^\n]*",
        r"\*?\s*ajude o canal[^\n]*",
        r"#publi\b[^\n]*",
        r"#ad\b[^\n]*",
        r"#publicidade\b[^\n]*",
        r"-?\s*anúncio\b[^\n]*",
        r"🫵🏼.*?(?:\n|$)",
        r"Convide Amigos.*?(?:\n|$)",
        r"https?://(?:chat\.whatsapp\.com|t\.me|grupos\.[^\s]+)[^\n]*",
    ]

    cleaned_text = text
    for pattern in patterns:
        cleaned_text = re.sub(pattern, "", cleaned_text, flags=re.IGNORECASE)

    cleaned_text = re.sub(r"\n\s*\n\s*\n", "\n\n", cleaned_text)
    return cleaned_text.strip()


# ---------- CONVERSÃO DE LINK DO MERCADO LIVRE (VIA ML_OFFER) ----------
async def convert_mercadolivre_link(product_url: str) -> dict | None:
    try:
        offer = await ml_offer.get_ml_offer(product_url)
        if not offer:
            if ml_offer.last_error:
                logger.error(f"❌ Erro no ml_offer: {ml_offer.last_error}")
            return None
        return offer
    except Exception as e:
        logger.error(f"❌ Exceção ao converter link do Mercado Livre no bot: {e}")
        return None


async def ml_worker() -> None:
    while True:
        job = await ml_state.queue.get()
        product_url: str = job["product_url"]
        future: asyncio.Future = job["future"]
        try:
            result = await convert_mercadolivre_link(product_url)
            if not future.cancelled():
                future.set_result(result)
        except Exception as e:
            if not future.cancelled():
                future.set_exception(e)
        finally:
            ml_state.queue.task_done()


async def enqueue_ml_conversion(product_url: str) -> dict | None:
    future: asyncio.Future = asyncio.get_running_loop().create_future()
    await ml_state.queue.put({"product_url": product_url, "future": future})
    return await future

async def handle_message(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await maybe_warn_cookie_expiration(update)

    raw_text = update.message.text or update.message.caption or ""

    shopee_links = [
        l.rstrip(".,;:!?)\n") for l in re.findall(SHOPEE_URL_PATTERN, raw_text)
    ]
    ml_links = [l.rstrip(".,;:!?)\n") for l in re.findall(ML_URL_PATTERN, raw_text)]

    if not shopee_links and not ml_links:
        await update.message.reply_text(
            "👋 Nenhum link válido da Shopee ou do Mercado Livre foi encontrado na mensagem.\n"
            "💡 Dica: Use /gerar para adicionar ofertas automáticas na fila!"
        )
        return

    cleaned_text = remove_affiliate_disclaimers(clean_corrupted_text(raw_text))

    if shopee_links:
        context.user_data["platform"] = "shopee"
        context.user_data["shopee_links"] = shopee_links
        context.user_data["raw_url"] = shopee_links[-1]
        context.user_data["multiple_ml"] = False
    else:
        context.user_data["platform"] = "mercadolivre"
        context.user_data["ml_links"] = ml_links
        context.user_data["raw_url"] = ml_links[-1]
        context.user_data["multiple_ml"] = len(ml_links) > 1

    context.user_data["raw_text"] = cleaned_text

    keyboard = InlineKeyboardMarkup(
        [
            [InlineKeyboardButton("🎟️ CUPOM", callback_data="type_cupom")],
            [
                InlineKeyboardButton(
                    "🏷️ OFERTA (Manter Texto)", callback_data="type_oferta"
                )
            ],
            [
                InlineKeyboardButton(
                    "🔗 LINK (Gerar Copy Automática)", callback_data="type_link"
                )
            ],
        ]
    )

    platform_name = (
        "Shopee" if context.user_data["platform"] == "shopee" else "Mercado Livre"
    )
    qtd_links = len(shopee_links) if shopee_links else len(ml_links)

    await update.message.reply_text(
        f"⚙️ <b>{qtd_links} Link(s) do {platform_name} identificado(s)!</b>\n\n"
        "Qual o tipo de mensagem que você deseja formatar?\n\n"
        "• <b>CUPOM:</b> Converte todos os links ou gera card especial de Central de Cupons.\n"
        "• <b>OFERTA:</b> Mantém seu texto original e substitui cada link pelo seu respectivo link de afiliado.\n"
        "• <b>LINK:</b> Gera copy completa via API (e inclui o link do cupom se houver mais de 1 link!).",
        reply_markup=keyboard,
        parse_mode="HTML",
    )


async def handle_callback(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()

    choice = query.data
    platform = context.user_data.get("platform", "shopee")
    raw_text = context.user_data.get("raw_text", "")
    raw_url = context.user_data.get("raw_url", "")
    multiple_ml = context.user_data.get("multiple_ml", False)

    if not raw_text or not raw_url:
        await query.edit_message_text(
            "⚠️ Os dados da mensagem expiraram. Envie o link novamente."
        )
        return

    await query.edit_message_text("⏳ Convertendo link(s) para afiliado...")
    clean_base_text = remove_affiliate_disclaimers(raw_text)

    media_url = None
    aff_link = None
    formatted_msg = ""
    category_tag = "Manual"

    # ================= MERCADO LIVRE =================
    if platform == "mercadolivre":
        ml_links = context.user_data.get("ml_links", [raw_url])

        if choice == "type_cupom":
            category_tag = "Cupom ML"
            url_map = {}
            for orig_l in ml_links:
                try:
                    res_link = await asyncio.to_thread(
                        create_affiliate_link_full, orig_l
                    )
                    conv = res_link.get("short_url") or res_link.get("long_url")
                    url_map[orig_l] = conv if conv else ML_DEFAULT_CUPON_LINK
                except Exception:
                    url_map[orig_l] = ML_DEFAULT_CUPON_LINK

            aff_link = url_map.get(raw_url, ML_DEFAULT_CUPON_LINK)

            if clean_base_text.strip() == raw_url.strip():
                formatted_msg = ShopeeAffiliateAPI.format_coupon_page_message(
                    aff_link, "Mercado Livre"
                )
            else:
                rewritten = escape_html(clean_base_text)
                for orig_l, new_l in url_map.items():
                    rewritten = rewritten.replace(escape_html(orig_l), new_l)
                formatted_msg = rewritten

        elif multiple_ml or len(ml_links) > 1:
            url_map = {}
            main_ml_result = None
            coupon_ml_links = []

            for orig_l in ml_links:
                res = await enqueue_ml_conversion(orig_l)
                if res and res.get("affiliate_url"):
                    url_map[orig_l] = res["affiliate_url"]
                    if res.get("current_price") is not None:
                        main_ml_result = res
                        media_url = res.get("image_url") or media_url
                    else:
                        coupon_ml_links.append(res["affiliate_url"])
                else:
                    try:
                        res_link = await asyncio.to_thread(
                            create_affiliate_link_full, orig_l
                        )
                        conv = res_link.get("short_url") or orig_l
                        url_map[orig_l] = conv
                        coupon_ml_links.append(conv)
                    except Exception:
                        url_map[orig_l] = orig_l

            if choice == "type_link" and main_ml_result:
                formatted_msg = ml_offer.format_product_message(
                    main_ml_result, main_ml_result.get("affiliate_url")
                )
                if coupon_ml_links:
                    c_lines = "\n".join(
                        f"🎟️ <b>Resgate o cupom aqui:</b> {cl}"
                        for cl in coupon_ml_links
                    )
                    formatted_msg = f"{formatted_msg}\n\n{c_lines}"
            else:
                new_text = escape_html(clean_base_text)
                for orig_l, new_l in url_map.items():
                    new_text = new_text.replace(escape_html(orig_l), new_l)

                free_notice = "\n\n🚚 <b>Aproveite o frete grátis e benefícios do Mercado Livre!</b>"
                if "frete grátis" not in new_text.lower():
                    new_text += free_notice
                formatted_msg = new_text

        else:
            ml_result = await enqueue_ml_conversion(raw_url)
            if not ml_result:
                await query.message.reply_text(
                    "❌ Não foi possível gerar o link de afiliado do Mercado Livre. Verifique se a sessão de login está válida."
                )
                return

            aff_link = ml_result.get("affiliate_url")
            media_url = ml_result.get("image_url")

            if choice == "type_link":
                formatted_msg = ml_offer.format_product_message(ml_result, aff_link)

"""
Lê um cURL (formato bash) salvo em curl.txt e atualiza o .env com:
  ML_AFFILIATE_COOKIE, ML_AFFILIATE_CSRF, ML_AFFILIATE_TAG

Uso:
  1. No navegador: F12 > Network > Fetch/XHR > gere um link no linkbuilder
  2. Botão direito na requisição 'createLink' > Copy > Copy as cURL (bash)
  3. Cole tudo em um arquivo chamado curl.txt (na pasta do projeto)
  4. python curl_to_env.py
"""

import os
import re
import sys

CURL_FILE = sys.argv[1] if len(sys.argv) > 1 else "curl.txt"
ENV_FILE = ".env"

if not os.path.exists(CURL_FILE):
    sys.exit(f"❌ Arquivo {CURL_FILE} não encontrado. Cole o cURL nele e rode de novo.")

with open(CURL_FILE, "r", encoding="utf-8") as f:
    text = f.read()

# Junta linhas quebradas com '\' (formato bash)
text = text.replace("\\\n", " ")

cookie = re.search(r"(?:-b|--cookie)\s+'([^']*)'", text)
csrf = re.search(r"-H\s+'x-csrf-token:\s*([^']+)'", text, re.I)
tag = re.search(r'"tag"\s*:\s*"([^"]+)"', text)

values = {}
if cookie:
    values["ML_AFFILIATE_COOKIE"] = f"'{cookie.group(1).strip()}'"
if csrf:
    values["ML_AFFILIATE_CSRF"] = csrf.group(1).strip()
if tag:
    values["ML_AFFILIATE_TAG"] = tag.group(1).strip()

missing = [
    k
    for k in ("ML_AFFILIATE_COOKIE", "ML_AFFILIATE_CSRF", "ML_AFFILIATE_TAG")
    if k not in values
]
if missing:
    print(f"⚠️ Não encontrei no cURL: {', '.join(missing)}")
    print("   Confira se copiou como 'cURL (bash)' e se é a requisição createLink.")

if not values:
    sys.exit("❌ Nada para atualizar.")

# Atualiza (ou acrescenta) as chaves no .env, preservando o resto
lines = []
if os.path.exists(ENV_FILE):
    with open(ENV_FILE, "r", encoding="utf-8") as f:
        lines = f.read().splitlines()

for key, val in values.items():
    new_line = f"{key}={val}"
    for i, line in enumerate(lines):
        if line.startswith(f"{key}="):
            lines[i] = new_line
            break
    else:
        lines.append(new_line)

with open(ENV_FILE, "w", encoding="utf-8") as f:
    f.write("\n".join(lines) + "\n")

for key, val in values.items():
    print(f"✅ {key} atualizado ({len(val)} caracteres)")

# O cURL contém a sessão: apaga o arquivo depois de usar
os.remove(CURL_FILE)
print(f"🗑️ {CURL_FILE} removido (continha dados da sessão).")

import sqlite3
from datetime import datetime, timezone, timedelta

DB_NAME = "database.db"
cuiaba_tz = timezone(timedelta(hours=-4))


def init_db():
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()

    cursor.execute("""
        CREATE TABLE IF NOT EXISTS config (
            key TEXT PRIMARY KEY,
            value TEXT
        )
    """)

    cursor.execute("""
        CREATE TABLE IF NOT EXISTS queue (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            store TEXT,
            product_url TEXT,
            formatted_message TEXT,
            media_url TEXT,
            media_type TEXT DEFAULT 'image', -- NOVO: 'image' ou 'video'
            priority INTEGER DEFAULT 0,
            status TEXT DEFAULT 'pending',
            target_date TEXT,
            scheduled_at TIMESTAMP,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            category TEXT DEFAULT 'Geral'
        )
    """)

    try:
        cursor.execute("ALTER TABLE queue ADD COLUMN media_type TEXT DEFAULT 'image'")
        conn.commit()
    except sqlite3.OperationalError:
        pass

    try:
        cursor.execute("ALTER TABLE queue ADD COLUMN category TEXT DEFAULT 'Geral'")
        conn.commit()
    except sqlite3.OperationalError:
        pass

    default_configs = {
        "start_time": "08:00",
        "end_time": "22:00",
        "interval_minutes": "30",
        "is_active": "1",
        "auto_refill_enabled": "1",  # 1 = Ativado (Opção A: abastece quando a fila zera)
        "auto_refill_count": "15",  # Quantidade gerada automaticamente por lote
        "auto_keywords": "",  # Palavras-chave opcionais separadas por vírgula
    }

    for key, val in default_configs.items():
        cursor.execute(
            "INSERT OR IGNORE INTO config (key, value) VALUES (?, ?)", (key, val)
        )

    conn.commit()
    conn.close()


def get_config(key: str) -> str:
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute("SELECT value FROM config WHERE key = ?", (key,))
    row = cursor.fetchone()
    conn.close()
    return row[0] if row else ""


def update_config(key: str, value: str):
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute(
        "INSERT OR REPLACE INTO config (key, value) VALUES (?, ?)", (key, value)
    )
    conn.commit()
    conn.close()


def is_offer_recent(
    product_url: str,
    item_id: int | str | None = None,
    days: int = 2,
    target_date: str | None = None,
) -> bool:
    """Verifica se o link ou itemId já foi adicionado nos últimos 'days' dias ou na data alvo."""
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()

    cutoff_date = (datetime.now(cuiaba_tz) - timedelta(days=days)).strftime("%Y-%m-%d")

    if target_date:
        cursor.execute(
            "SELECT product_url FROM queue WHERE target_date >= ? OR target_date = ?",
            (cutoff_date, target_date),
        )
    else:
        cursor.execute(
            "SELECT product_url FROM queue WHERE target_date >= ?",
            (cutoff_date,),
        )

    existing_urls = [row[0] for row in cursor.fetchall() if row[0]]
    conn.close()

    item_str = str(item_id) if item_id else None
    for url in existing_urls:
        if url == product_url:
            return True
        if item_str and item_str in url:
            return True

    return False


def add_to_queue(
    store: str,
    product_url: str,
    message: str,
    media_url: str = None,
    media_type: str = "image",
    priority: int = 0,
    target_date: str = None,
    category: str = "Geral",
):
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    now_cuiaba = datetime.now(cuiaba_tz).strftime("%Y-%m-%d %H:%M:%S")
    if not target_date:
        target_date = datetime.now(cuiaba_tz).strftime("%Y-%m-%d")

    cursor.execute(
        """
        INSERT INTO queue (store, product_url, formatted_message, media_url, media_type, priority, status, target_date, scheduled_at, created_at, category)
        VALUES (?, ?, ?, ?, ?, ?, 'pending', ?, ?, ?, ?)
    """,
        (
            store,
            product_url,
            message,
            media_url,
            media_type,
            priority,
            target_date,
            now_cuiaba,
            now_cuiaba,
            category or "Geral",
        ),
    )
    conn.commit()
    conn.close()


def get_next_pending_offer(target_date: str):
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute(
        """
        SELECT id, store, product_url, formatted_message, media_url, priority, COALESCE(media_type, 'image')
        FROM queue
        WHERE status = 'pending' AND target_date = ?
        ORDER BY priority DESC, id ASC
        LIMIT 1
    """,
        (target_date,),
    )
    row = cursor.fetchone()
    conn.close()
    return row


def mark_offer_as_sent(offer_id: int):
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute("UPDATE queue SET status = 'sent' WHERE id = ?", (offer_id,))
    conn.commit()
    conn.close()


def delete_offer(offer_id: int):
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute("DELETE FROM queue WHERE id = ?", (offer_id,))
    conn.commit()
    conn.close()

# ---------------------------------------------------------------------------

# ---------------------------------------------------------------------------
def _unwrap_verification(url: str) -> str | None:
    """O ML às vezes redireciona para /gz/account-verification?go=<url real>."""
    parsed = urlparse(url)
    if "account-verification" in parsed.path:
        go = (parse_qs(parsed.query).get("go") or [None])[0]
        if go:
            return go + (("#" + parsed.fragment) if parsed.fragment else "")
    return None


def _has_product_id(url: str) -> bool:
    p = urlparse(url)
    pdp = " ".join(parse_qs(p.query).get("pdp_filters", []))
    return bool(re.search(r"MLBU?-?\d+", f"{p.path} {p.fragment} {pdp}", re.I))


def expand_url(url: str, max_hops: int = 8) -> str:
    """Segue os redirecionamentos 1 a 1 (sem baixar a página) até achar a URL do produto."""
    current = url.strip()
    for _ in range(max_hops):
        unwrapped = _unwrap_verification(current)
        if unwrapped:
            return unwrapped

        if _has_product_id(current):
            return current

        base, frag = urldefrag(current)
        try:
            r = requests.get(
                base,
                headers={"User-Agent": USER_AGENT},
                allow_redirects=False,
                timeout=15,
                stream=True,
            )
            location = r.headers.get("Location")
            r.close()
        except Exception as e:
            print(f"   ⚠️ Erro ao expandir {base}: {e}")
            return current

        if not location:
            return current

        nxt = urljoin(base, location)
        if "#" not in nxt and frag:
            nxt += "#" + frag
        current = nxt
    return current


# ---------------------------------------------------------------------------
# 2. Limpar link (remove rastreio e parâmetros de outros afiliados)
# ---------------------------------------------------------------------------
def clean_url(url: str) -> str:
    parsed = urlparse(url)
    qs = parse_qs(parsed.query)
    frag = parse_qs(parsed.fragment)

    query = urlencode({k: qs[k][0] for k in KEEP_QUERY if k in qs})
    fragment = urlencode({k: frag[k][0] for k in KEEP_FRAGMENT if k in frag})
    return parsed._replace(query=query, fragment=fragment).geturl()

def _collect_urls(obj, found: list):
    if isinstance(obj, str):
        if obj.startswith("http"):
            found.append(obj)
    elif isinstance(obj, dict):
        for v in obj.values():
            _collect_urls(v, found)
    elif isinstance(obj, list):
        for v in obj:
            _collect_urls(v, found)


def _extract_link(data, original: str) -> str:
    urls = data.get("urls") if isinstance(data, dict) else None
    if urls and isinstance(urls, list) and isinstance(urls[0], dict):
        first = urls[0]
        if first.get("error_code"):
            raise ValueError(
                f"O ML recusou a URL ({first.get('message')}, "
                f"error_code={first.get('error_code')}): {first.get('origin_url')}"
            )
        if first.get("short_url"):
            return first["short_url"]

    found: list = []
    _collect_urls(data, found)
    candidates = [u for u in found if u.split("#")[0] != original.split("#")[0]]

    for u in candidates:
        if "meli.la" in u or "/sec/" in u:
            return u
    if candidates:
        return candidates[0]
    raise ValueError(f"Não encontrei o link de afiliado na resposta: {str(data)[:300]}")


def create_affiliate_link_full(
    product_url: str, tag: str | None = None, debug: bool = False
) -> dict:
    """Cria o link de afiliado e devolve {'short_url', 'long_url', 'text'}."""
    cookie = os.environ.get("ML_AFFILIATE_COOKIE")
    csrf = os.environ.get("ML_AFFILIATE_CSRF")
    tag = tag or os.environ.get("ML_AFFILIATE_TAG")
    apikey = os.environ.get("ML_AFFILIATE_APIKEY", "mude-me")

    if not (cookie and csrf and tag):
        raise RuntimeError(
            "Defina ML_AFFILIATE_COOKIE, ML_AFFILIATE_CSRF e ML_AFFILIATE_TAG no .env"
        )

    headers = {
        "accept": "application/json, text/plain, */*",
        "accept-language": "pt-BR,pt;q=0.9,en-US;q=0.8,en;q=0.7",
        "apikey": apikey,
        "content-type": "application/json",
        "origin": "https://www.mercadolivre.com.br",
        "referer": REFERER,
        "user-agent": USER_AGENT,
        "x-csrf-token": csrf,
        "cookie": cookie,
    }

    resp = requests.post(
        CREATE_LINK_URL,
        headers=headers,
        json={"urls": [product_url], "tag": tag},
        timeout=20,
    )

    if debug:
        print(f"   createLink status={resp.status_code} body={resp.text[:500]}")

    if resp.status_code in (401, 403):
        raise AffiliateSessionExpired(
            f"Sessão do afiliado recusada (HTTP {resp.status_code}). Atualize cookie/CSRF no .env."
        )
    resp.raise_for_status()

    data = resp.json()
    short_url = _extract_link(
        data, product_url
    )  # valida e levanta erro claro se o ML recusar

    first = {}
    if isinstance(data, dict) and isinstance(data.get("urls"), list) and data["urls"]:
        first = data["urls"][0] if isinstance(data["urls"][0], dict) else {}

    return {
        "short_url": short_url,
        "long_url": first.get("long_url"),
        "text": first.get("text"),
    }


def create_affiliate_link(
    product_url: str, tag: str | None = None, debug: bool = False
) -> str:
    return create_affiliate_link_full(product_url, tag=tag, debug=debug)["short_url"]

def ensure_product_url(url: str):
    """Garante que a URL aponta para um produto (e não para perfil/vitrine de afiliado)."""
    if not _has_product_id(url):
        raise ValueError(

def resolve_product_url(url: str, debug: bool = False):
    """Expande o link e devolve (expandido, url_do_produto, info).

    Para links de outros afiliados (que abrem /social/<perfil>), lê o produto
    compartilhado dentro da própria página. Nos demais casos, info = None.
    """
    expanded = expand_url(url)
    info = None
    if "/social/" in urlparse(expanded).path:
        info = fetch_social_product(expanded, debug=debug)
        product_url = info["product_url"]
    else:
        product_url = expanded
    return expanded, product_url, info

def _normalize_text(text: str) -> str:
    if not text:
        return ""
    nfkd = unicodedata.normalize("NFKD", text)
    return "".join(c for c in nfkd if not unicodedata.combining(c)).lower().strip()


def parse_csv_list(raw_csv: str | None) -> list[str]:
    if not raw_csv:
        return []
    return [item.strip() for item in raw_csv.split(",") if item.strip()]


def extract_ml_item_key(url: str, metadata: dict | None = None) -> str:
    if metadata:
        for k in ("product_id", "user_product_id", "id", "pid"):
            val = metadata.get(k)
            if val:
                return str(val).upper()

    if not url:
        return ""

    m_cat = re.search(r"/p/(MLB\d+)", url, re.IGNORECASE)
    if m_cat:
        return m_cat.group(1).upper()

    m_up = re.search(r"/up/(MLBU\d+)", url, re.IGNORECASE)
    if m_up:
        return m_up.group(1).upper()

    m_item = re.search(r"(MLB-?\d+)", url, re.IGNORECASE)
    if m_item:
        return m_item.group(1).replace("-", "").upper()

    return clean_url(url)


def _extract_nordic_ctx(html_text: str) -> dict | None:
    """Extrai o JSON de _n.ctx.r presente no HTML do Mercado Livre."""
    for marker in ("_n.ctx.r=", "_n.ctx.r ="):
        idx = html_text.find(marker)
        if idx != -1:
            start = html_text.find("{", idx)
            if start != -1:
                try:
                    decoder = json.JSONDecoder()
                    data, _ = decoder.raw_decode(html_text, start)
                    return data
                except Exception as e:
                    logger.warning(f"Falha ao decodificar _n.ctx.r: {e}")
    return None


def _parse_polycard(
    polycard: dict, source_type: str = "offers", category_name: str = ""
) -> dict | None:
    """Converte um objeto polycard do JSON do ML em dicionário padronizado de oferta."""
    if not isinstance(polycard, dict):
        return None

    metadata = polycard.get("metadata") or {}
    raw_url = metadata.get("url") or ""
    if not raw_url:
        return None

    url_params = metadata.get("url_params") or ""
    full_url = raw_url if raw_url.startswith("http") else f"https://{raw_url}"
    cleaned_url = clean_url(full_url)
    if "searchVariation" in url_params and "?" not in cleaned_url:
        product_url_full = f"{cleaned_url}{url_params}"
    else:
        product_url_full = cleaned_url

    item_key = extract_ml_item_key(cleaned_url, metadata)

    pic_id = None
    pics_list = (polycard.get("pictures") or {}).get("pictures") or []
    if pics_list and isinstance(pics_list, list):
        pic_id = pics_list[0].get("id")
    image_url = (
        f"https://http2.mlstatic.com/D_Q_NP_2X_{pic_id}-AB.webp" if pic_id else None
    )

    title = None
    seller = None
    current_price = None
    original_price = None
    regular_price = None
    pix_price = None
    price_note = None
    discount_label = None
    installments_text = None
    installments_no_interest = None
    has_shipping_info = True

    for comp in polycard.get("components") or []:
        if not isinstance(comp, dict):
            continue
        c_type = comp.get("type")

        if c_type == "title":
            title = (comp.get("title") or {}).get("text")

        elif c_type == "seller":
            seller_obj = comp.get("seller") or {}
            seller_text = seller_obj.get("text") or ""
            for v in seller_obj.get("values") or []:
                if v.get("key") == "label":
                    seller = (v.get("label") or {}).get("text")
                    break
            if not seller and seller_text:
                clean_s = re.sub(r"\{[^}]+\}", "", seller_text).strip()
                if clean_s:
                    seller = clean_s

        elif c_type == "price":
            p_obj = comp.get("price") or {}
            cur_obj = p_obj.get("current_price") or {}
            current_price = cur_obj.get("value")

            for pl in p_obj.get("price_labels") or []:
                for val in pl.get("values") or []:
                    if val.get("key") == "previous_price":
                        original_price = (val.get("price") or {}).get("value")

            disc_poly = p_obj.get("discount_polylabel") or {}
            for val in disc_poly.get("values") or []:
                pill_text = (val.get("pill") or {}).get("text")
                if pill_text:
                    discount_label = pill_text.strip()

            unit_desc = (
                (p_obj.get("unit_description") or {}).get("text") or ""
            ).strip()

            inst_obj = p_obj.get("installments") or {}
            inst_template = inst_obj.get("text") or ""
            inst_values = inst_obj.get("values") or []

            if "pix" in unit_desc.lower():
                pix_price = current_price
                price_note = "no Pix"
                regular_price = current_price
                for v in inst_values:
                    if v.get("key") == "price_total":
                        regular_price = (v.get("price") or {}).get(
                            "value"
                        ) or current_price
            else:
                regular_price = current_price
                if unit_desc:
                    price_note = unit_desc

                if inst_template and "outros meios" not in inst_template.lower():
                    rendered = inst_template
                    for v in inst_values:
                        k = v.get("key")
                        if v.get("type") == "price":
                            val_num = (v.get("price") or {}).get("value")
                            if val_num is not None:
                                brl_str = (
                                    f"R$ {val_num:,.2f}".replace(",", "X")
                                    .replace(".", ",")
                                    .replace("X", ".")
                                )
                                rendered = rendered.replace(f"{{{k}}}", brl_str)
                        elif v.get("type") == "label":
                            lbl_str = (v.get("label") or {}).get("text") or ""
                            rendered = rendered.replace(f"{{{k}}}", lbl_str)
                    installments_text = re.sub(r"\s+", " ", rendered).strip()
                    installments_no_interest = bool(inst_obj.get("no_interest"))

        elif c_type == "shipping_v2":
            if comp.get("shipping_v2") is None and source_type == "best_sellers":
                has_shipping_info = False

    if source_type == "best_sellers" and not has_shipping_info:
        return None

    if not title or current_price is None:
        return None

    return {
        "item_key": item_key,
        "product_url": product_url_full,
        "clean_url": cleaned_url,
        "affiliate_url": None,
        "image_url": image_url,
        "title": title,
        "current_price": float(current_price),
        "original_price": float(original_price) if original_price else None,
        "regular_price": (
            float(regular_price) if regular_price else float(current_price)
        ),
        "pix_price": float(pix_price) if pix_price else None,
        "price_note": price_note,
        "discount_label": discount_label,
        "installments_text": installments_text,
        "installments_no_interest": installments_no_interest,
        "seller": seller,
        "source_type": source_type,
        "category_name": category_name or "Mercado Livre",
    }


def _find_carousels_recursively(node, found_list: list):
    """Busca recursivamente todos os carrosséis de categorias dentro do JSON de Mais Vendidos."""
    if isinstance(node, dict):
        # Se este dicionário tem 'title' e 'polycards' (ou 'items' -> 'result' -> 'polycards')
        polycards = (
            ((node.get("items") or {}).get("result") or {}).get("polycards")
            if isinstance(node.get("items"), dict)
            else None
        )
        if polycards and isinstance(polycards, list):
            found_list.append(
                {
                    "title": node.get("title") or "",
                    "polycards": polycards,
                }
            )
        else:
            for v in node.values():
                if isinstance(v, (dict, list)):
                    _find_carousels_recursively(v, found_list)
    elif isinstance(node, list):
        for item in node:
            if isinstance(item, (dict, list)):
                _find_carousels_recursively(item, found_list)


def _find_offer_cards_recursively(node, found_cards: list):
    """Busca recursivamente todos os polycards dentro do JSON da página de Ofertas."""
    if isinstance(node, dict):
        if (
            "metadata" in node
            and "components" in node
            and isinstance(node["metadata"], dict)
        ):
            found_cards.append(node)
            return
        for v in node.values():
            if isinstance(v, (dict, list)):
                _find_offer_cards_recursively(v, found_cards)
    elif isinstance(node, list):
        for item in node:
            if isinstance(item, (dict, list)):
                _find_offer_cards_recursively(item, found_cards)


def fetch_ml_best_sellers_candidates(
    excluded_categories: list[str] | None = None,
) -> tuple[list[dict], list[str], dict[str, int]]:
    """
    Coleta os produtos da página Mais Vendidos do ML filtrando as categorias excluídas.
    Retorna: (lista_candidatos, categorias_bloqueadas, contagem_por_categoria_aceita)
    """
    if excluded_categories is None:
        excluded_categories = parse_csv_list(
            get_config("ml_excluded_categories_bestsellers") or "Games, Informática"
        )
    excluded_norm = [_normalize_text(c) for c in excluded_categories if c.strip()]

    session = requests.Session()
    resp = session.get(
        DEFAULT_ML_MAIS_VENDIDOS_URL, headers=_build_ml_headers(), timeout=20
    )
    resp.raise_for_status()

    ctx = _extract_nordic_ctx(resp.text)
    if not ctx:
        logger.warning("⚠️ _n.ctx.r não encontrado em Mais Vendidos.")
        return [], [], {}

    carousels = []
    _find_carousels_recursively(ctx, carousels)

    candidates: list[dict] = []
    blocked_cats: list[str] = []
    accepted_cats: dict[str, int] = {}

    for car in carousels:
        raw_title = car.get("title") or ""
        cat_clean = re.sub(
            r"^Mais vendidos em\s+", "", raw_title, flags=re.IGNORECASE
        ).strip()
        if not cat_clean:
            cat_clean = "Geral"
        cat_norm = _normalize_text(cat_clean)

        if any(excl in cat_norm or cat_norm in excl for excl in excluded_norm if excl):
            if cat_clean not in blocked_cats:
                blocked_cats.append(cat_clean)
            continue

        count_ok = 0
        for card in car.get("polycards") or []:
            parsed = _parse_polycard(
                card,
                source_type="best_sellers",
                category_name=f"ML: {cat_clean}",
            )
            if parsed:
                candidates.append(parsed)
                count_ok += 1

        if count_ok > 0:
            accepted_cats[cat_clean] = accepted_cats.get(cat_clean, 0) + count_ok

    return candidates, blocked_cats, accepted_cats


def fetch_ml_offers_candidates(
    offers_url: str | None = None,
    excluded_terms: list[str] | None = None,
    max_pages: int = 2,
) -> list[dict]:
    """
    Coleta produtos da página de Ofertas (ex: Menos de R$ 100), buscando até `max_pages` páginas.
    """
    base_url = (
        offers_url or get_config("ml_offers_url") or DEFAULT_ML_OFERTAS_URL
    ).strip()
    base_url_clean = base_url.split("#")[0]

    if excluded_terms is None:
        excluded_terms = parse_csv_list(
            get_config("ml_excluded_categories_offers") or ""
        )
    excluded_norm = [_normalize_text(t) for t in excluded_terms if t.strip()]

    session = requests.Session()
    headers = _build_ml_headers()
    candidates: list[dict] = []
    seen_keys: set[str] = set()

    for page_num in range(1, max_pages + 1):
        sep = "&" if "?" in base_url_clean else "?"
        page_url = (
            base_url_clean if page_num == 1 else f"{base_url_clean}{sep}page={page_num}"
        )
        try:
            resp = session.get(page_url, headers=headers, timeout=20)
            if resp.status_code != 200:
                break
            ctx = _extract_nordic_ctx(resp.text)
            if not ctx:
                break

            raw_cards = []
            _find_offer_cards_recursively(ctx, raw_cards)

            for card in raw_cards:
                parsed = _parse_polycard(
                    card,
                    source_type="offers",
                    category_name="ML: Ofertas (< R$100)",
                )
                if not parsed:
                    continue
                if parsed["item_key"] in seen_keys:
                    continue
                seen_keys.add(parsed["item_key"])

                if excluded_norm:
                    t_norm = _normalize_text(parsed["title"])
                    u_norm = _normalize_text(parsed["clean_url"])
                    if any(ex in t_norm or ex in u_norm for ex in excluded_norm if ex):
                        continue

                candidates.append(parsed)
        except Exception as e:
            logger.warning(f"Erro ao ler página {page_num} de Ofertas ML: {e}")
            break

    return candidates


def simulate_ml_offers_selection(count: int = 10) -> dict:
    """
    Executa a busca completa do Mercado Livre em modo SIMULAÇÃO (Dry-Run / Laboratório).
    Filtra categorias excluídas, aplica deduplicação semântica de 7 dias e sorteia
    respeitando os 30% Mais Vendidos e 70% Ofertas (< R$ 100), SEM gastar conversão de link.
    """
    t0 = time.perf_counter()

    best_pct = int(get_config("ml_best_sellers_pct") or "30")
    best_pct = max(0, min(100, best_pct))
    dedup_days = int(get_config("ml_history_days") or "7")
    dedup_threshold = float(get_config("dedup_similarity_threshold") or "60")

    target_best = round(count * (best_pct / 100.0))
    if best_pct > 0 and count >= 2 and target_best == 0:
        target_best = 1
    target_offers = max(0, count - target_best)

    best_raw, blocked_cats, accepted_cats = fetch_ml_best_sellers_candidates()
    offers_raw = fetch_ml_offers_candidates(max_pages=2)

    preloaded_profiles = offer_comparator.load_recent_queue_profiles(days=dedup_days)

    random.shuffle(best_raw)
    random.shuffle(offers_raw)

    accepted_titles: list[str] = []
    seen_keys: set[str] = set()
    skipped_duplicates = 0

    def _pick_from_pool(pool: list[dict], needed: int) -> list[dict]:
        nonlocal skipped_duplicates
        picked = []
        for item in pool:
            if len(picked) >= needed:
                break
            ik = item["item_key"]
            if ik in seen_keys:
                continue
            seen_keys.add(ik)

            is_dup, reason = offer_comparator.is_product_recently_used_or_blocked(
                candidate_title=item["title"],
                candidate_url=item["clean_url"],
                days=dedup_days,
                threshold=dedup_threshold,
                extra_accepted_titles=accepted_titles,
                preloaded_db_profiles=preloaded_profiles,
            )
            if is_dup:
                skipped_duplicates += 1
                continue

            accepted_titles.append(item["title"])
            item_preview = {
                **item,
                "affiliate_url": "https://meli.la/PREVIA_SIMULACAO",
            }
            item["preview_message"] = ml_offer.format_product_message(item_preview)
            picked.append(item)
        return picked

    chosen_best = _pick_from_pool(best_raw, target_best)
    chosen_offers = _pick_from_pool(offers_raw, target_offers)

    # Completa caso uma das fontes tenha menos itens
    total_got = len(chosen_best) + len(chosen_offers)
    if total_got < count:
        missing = count - total_got
        extra = _pick_from_pool(offers_raw + best_raw, missing)
        chosen_offers.extend(extra)

    # Intercala os itens
    final_items = []
    cb = list(chosen_best)
    co = list(chosen_offers)
    while cb or co:
        for _ in range(2):
            if co:
                final_items.append(co.pop(0))
        if cb:
            final_items.append(cb.pop(0))

    elapsed_ms = round((time.perf_counter() - t0) * 1000, 1)

    return {
        "elapsed_ms": elapsed_ms,
        "blocked_categories": blocked_cats,
        "accepted_categories": accepted_cats,
        "raw_best_count": len(best_raw),
        "raw_offers_count": len(offers_raw),
        "skipped_duplicates": skipped_duplicates,
        "chosen_best_count": len(chosen_best),
        "chosen_offers_count": len(chosen_offers),
        "items": final_items,
    }


def convert_and_enqueue_ml_item(
    product_url: str,
    target_date: str,
    category_name: str = "ML: Automático",
    priority: int = 0,
    fallback_card_data: dict | None = None,
) -> bool:
    """
    Converte 1 produto aprovado do Mercado Livre em link oficial de afiliado (meli.la),
    formata a mensagem com a função única `ml_offer.format_product_message` e salva na fila.
    """
    try:
        offer = ml_offer._build_offer_sync(product_url)
        if not offer or not offer.get("affiliate_url"):
            return False

        # Se algum dado veio mais rico no card da listagem (ex: discount_label), preserva como fallback
        if fallback_card_data:
            for k in (
                "title",
                "current_price",
                "original_price",
                "pix_price",
                "discount_label",
                "image_url",
            ):
                if not offer.get(k) and fallback_card_data.get(k):
                    offer[k] = fallback_card_data[k]

        msg = ml_offer.format_product_message(offer, offer["affiliate_url"])
        media_url = offer.get("image_url")

        add_to_queue(
            store="Mercado Livre",
            product_url=clean_url(product_url),
            message=msg,
            media_url=media_url,
            media_type="image",
            priority=priority,
            target_date=target_date,
            category=category_name,
        )
        return True
    except Exception as e:
        logger.error(f"Erro ao converter e enfileirar item ML ({product_url}): {e}")
        return False


def generate_mixed_auto_offers_to_queue(
    total_count: int, target_date: str, keywords_csv: str = ""
) -> dict:
    """
    Orquestrador de Geração Automática:
    - Se `ml_auto_enabled == '1'`, divide o lote em 60% Shopee e 40% Mercado Livre (configurável).
    - Se `ml_auto_enabled == '0'` (modo validação), gera 100% Shopee no automático enquanto você valida o ML na aba Laboratório.
    """
    ml_enabled = get_config("ml_auto_enabled") == "1"
    shopee_pct = int(get_config("mix_shopee_pct") or "60") if ml_enabled else 100
    shopee_pct = max(0, min(100, shopee_pct))

    shopee_target = round(total_count * (shopee_pct / 100.0))
    ml_target = max(0, total_count - shopee_target)

    ml_added = 0
    if ml_target > 0:
        sim = simulate_ml_offers_selection(count=ml_target + 3)
        for cand in sim["items"]:
            if ml_added >= ml_target:
                break
            ok = convert_and_enqueue_ml_item(
                product_url=cand["product_url"],
                target_date=target_date,
                category_name=cand.get("category_name", "ML: Automático"),
                priority=0,
                fallback_card_data=cand,
            )
            if ok:
                ml_added += 1

    # Compensa com Shopee se faltou algum item do ML
    shopee_actual_target = shopee_target + max(0, ml_target - ml_added)
    shopee_added = 0

    app_id = os.environ.get("SHOPEE_APP_ID")
    app_secret = os.environ.get("SHOPEE_APP_SECRET")
    if shopee_actual_target > 0 and app_id and app_secret:
        api = ShopeeAffiliateAPI(app_id, app_secret)
        shopee_added = api.generate_auto_offers_to_queue(
            count=shopee_actual_target,
            target_date=target_date,
            keywords_csv=keywords_csv,
        )

    return {
        "total_added": shopee_added + ml_added,
        "shopee_added": shopee_added,
        "ml_added": ml_added,
    }

"""
Monta a oferta do Mercado Livre sem navegador e sem API oficial,
usando apenas a conversão interna (createLink) e a leitura do HTML (_n.ctx.r).

Retorna:
    affiliate_url, image_url, title, current_price, original_price,
    regular_price, pix_price, discount_label, installments_text,
    installments_no_interest, seller, item_id, product_url
"""

import asyncio
import html
import logging
import requests

from ml_affiliate import (
    AffiliateSessionExpired,
    clean_url,
    create_affiliate_link_full,
    ensure_product_url,
    resolve_product_url,
)

logger = logging.getLogger(__name__)

# Último erro (texto) para o bot poder avisar, ex.: sessão do afiliado expirada
last_error: str | None = None


def _build_offer_sync(raw_url: str, debug: bool = False) -> dict:
    # 1. Expande o link. Se for link de outro afiliado (/social/), já traz 'info' completo!
    _, product_url, info = resolve_product_url(raw_url, debug=debug)

    # 2. Limpa a URL do produto e cria o seu link de afiliado (short_url + long_url)
    cleaned = clean_url(product_url)
    ensure_product_url(cleaned)

    link = create_affiliate_link_full(cleaned, debug=debug)
    affiliate_url = link["short_url"]

    # 3. Se o link enviado era um link direto de produto (info ainda é None),
    #    buscamos os dados na página do link gerado (short_url abre a página /social/ com o card!)
    if not info or info.get("price") is None:
        target_to_read = affiliate_url or link.get("long_url") or cleaned
        try:
            info = fetch_social_product(target_to_read, debug=debug)
        except (SocialPageBlocked, ValueError, requests.RequestException) as e:
            logger.warning(f"Não consegui ler os dados via {target_to_read}: {e}")
            if link.get("long_url") and link["long_url"] != target_to_read:
                try:
                    info = fetch_social_product(link["long_url"], debug=debug)
                except Exception:
                    pass

    info = info or {}
    price = info.get("price")
    pix = info.get("pix_price")

    return {
        # --- mesmas chaves esperadas pelo bot.py ---
        "affiliate_url": affiliate_url,
        "image_url": info.get("thumbnail"),
        "title": info.get("title"),
        "current_price": pix if pix is not None else price,
        "original_price": info.get("original_price"),
        "rating": None,
        "reviews_count": None,
        "sales": None,
        # --- detalhes de preço (Pix, Normal e Parcelamento) ---
        "regular_price": price,
        "pix_price": pix,
        "price_note": "no Pix" if pix is not None else None,
        "discount_label": info.get("discount_label"),
        "installments_text": info.get("installments_text"),
        "installments_no_interest": info.get("installments_no_interest"),
        "seller": info.get("seller"),
        "item_id": info.get("itemId"),
        "product_url": cleaned,
    }


async def get_ml_offer(product_url: str, debug: bool = False) -> dict | None:
    """Versão assíncrona (roda o código bloqueante em outra thread)."""
    global last_error
    last_error = None
    try:
        return await asyncio.to_thread(_build_offer_sync, product_url, debug)
    except AffiliateSessionExpired as e:
        last_error = str(e)
        logger.error(f"❌ {e}")
    except Exception as e:
        last_error = str(e)
        logger.error(f"❌ Erro ao montar a oferta do Mercado Livre: {e}")
    return None


def _brl(value) -> str:
    return "R$ " + f"{value:,.2f}".replace(",", "X").replace(".", ",").replace("X", ".")


def format_product_message(offer: dict, link: str | None = None) -> str:
    """Função única e centralizada para formatar a mensagem do Mercado Livre (Telegram, Portal e WhatsApp)."""
    aff_link = link or offer.get("affiliate_url") or ""
    raw_title = offer.get("title")
    title = html.escape(str(raw_title)) if raw_title else None

    header_title = (
        f"🔥 <b>{title}</b>\n\n"
        if title
        else "🔥 <b>PRODUTO EM DESTAQUE NO MERCADO LIVRE</b>\n\n"
    )

    cur = offer.get("current_price")
    orig = offer.get("original_price")
    pix = offer.get("pix_price")
    reg = offer.get("regular_price")
    discount_label = (offer.get("discount_label") or "").strip()
    inst_text = offer.get("installments_text")

    # --- 1. Destaque do Desconto (% OFF no Pix / Normal) ---
    is_pix_deal = bool(pix or (discount_label and "pix" in discount_label.lower()))
    discount_line = ""

    if discount_label:
        label_clean = html.escape(discount_label.replace("!", "").strip())
        if "pix" in label_clean.lower():
            discount_line = f"🔥 <b>{label_clean}</b>\n\n"
        elif is_pix_deal:
            discount_line = f"🔥 <b>{label_clean} no Pix</b>\n\n"
        else:
            discount_line = f"🔥 <b>{label_clean}</b>\n\n"
    elif orig and cur and orig > cur:
        percent_off = round(((orig - cur) / orig) * 100)
        if percent_off > 0:
            if is_pix_deal:
                discount_line = f"🔥 <b>{percent_off}% OFF no Pix</b>\n\n"
            else:
                discount_line = f"🔥 <b>{percent_off}% OFF aplicado no item!</b>\n\n"

    # --- 2. Preço e Parcelamento ---
    price_line = ""
    note = f" {offer['price_note']}" if offer.get("price_note") else ""

    if cur is not None:
        if orig and orig > cur:
            price_line += (
                f"💰 <b>Preço:</b> de {_brl(orig)} por <b>{_brl(cur)}</b>{note}\n"
            )
        else:
            price_line += f"💰 <b>Preço:</b> <b>{_brl(cur)}</b>{note}\n"

        if pix is not None and reg is not None and reg != pix:
            price_line += f"💳 ou <b>{_brl(reg)}</b> em outros meios\n"

        if inst_text:
            price_line += f"💳 Parcelamento: <b>{html.escape(str(inst_text))}</b>\n"

        price_line += "\n"

    # --- 3. Loja ---
    seller_line = ""
    if offer.get("seller"):
        seller_raw = offer["seller"].removeprefix("Por ").strip()

        # Se já tiver "Oficial" ou "Loja", tratamos para ficar elegante
        if "oficial" in seller_raw.lower():
            seller_clean = html.escape(seller_raw)
            seller_line = f"🏪 Loja: <b>{seller_clean} no Mercado Livre</b>\n\n"
        else:
            seller_clean = html.escape(seller_raw)
            seller_line = f"🏪 Loja: <b>{seller_clean} no Mercado Livre</b>\n\n"
    else:
        seller_line = "🏪 Loja: <b>Mercado Livre</b>\n\n"

    # --- 4. Frete e Link de Ação ---
    free_shipping = "🚚 <b>Aproveite frete grátis (conforme regras do app) e benefícios do Mercado Livre!</b>\n\n"
    cta_line = f"👇 <b>Clique no link abaixo para garantir o seu:</b>\n👉 {aff_link}"

    return f"{header_title}{discount_line}{price_line}{seller_line}{free_shipping}{cta_line}"


def preview_message(offer: dict) -> str:
    """Alias que reaproveita a função única format_product_message."""

def _brl_str(val: float) -> str:
    return "R$ " + f"{val:,.2f}".replace(",", "X").replace(".", ",").replace("X", ".")


def _extract_ctx(html: str) -> dict | None:
    m = re.search(r"_n\.ctx\.r\s*=\s*", html)
    if not m:
        return None
    try:
        obj, _ = json.JSONDecoder().raw_decode(html[m.end() :])
        return obj
    except ValueError:
        return None


def _walk(obj):
    if isinstance(obj, dict):
        yield obj
        for v in obj.values():
            yield from _walk(v)
    elif isinstance(obj, list):
        for v in obj:
            yield from _walk(v)


def _find_featured_polycard(data: dict, shared_ids: list[str]) -> dict | None:
    """Busca primeiro o polycard dentro do componente 'card-featured' ou pelo shared_id."""
    # 1. Procura direto no componente de destaque 'card-featured'
    for comp in data.get("components") or []:
        if isinstance(comp, dict) and comp.get("id") == "card-featured":
            pcs = (
                (comp.get("recommendation_data") or {}).get("recommendation_info") or {}
            ).get("polycards") or []
            if pcs and isinstance(pcs[0], dict):
                return pcs[0]

    # 2. Procura em todos os polycards comparando id, user_product_id e product_id
    all_cards = []
    for node in _walk(data):
        pcs = node.get("polycards")
        if isinstance(pcs, list):
            all_cards.extend(c for c in pcs if isinstance(c, dict))

    for sid in shared_ids:
        for c in all_cards:
            meta = c.get("metadata") or {}
            if sid in (
                meta.get("id"),
                meta.get("user_product_id"),
                meta.get("product_id"),
                meta.get("pid"),
            ):
                return c

    return all_cards[0] if all_cards else None


def _components(card: dict) -> dict:
    out = {}
    for c in card.get("components") or []:
        if isinstance(c, dict) and c.get("type"):
            out[c["type"]] = c.get(c["type"]) or {}
    return out


def _meta_tag(data: dict, tag_type: str) -> str | None:
    for tag in (data.get("header") or {}).get("meta_tags") or []:
        if isinstance(tag, dict) and tag.get("type") == tag_type:
            return tag.get("content")
    return None


def _format_interpolated_text(raw_text: str, values_list: list) -> str:
    """Substitui placeholders como {price_total} ou {0} pelos valores reais do JSON."""
    if not raw_text:
        return ""
    result = raw_text
    for item in values_list or []:
        if not isinstance(item, dict):
            continue
        key = item.get("key")
        if not key:
            continue
        val_str = ""
        if item.get("type") == "price" and isinstance(item.get("price"), dict):
            v = item["price"].get("value")
            if v is not None:
                val_str = _brl_str(float(v))
        elif "text" in item:
            val_str = str(item["text"])
        if val_str:
            result = result.replace(f"{{{key}}}", val_str)
    return re.sub(r"\s+", " ", result).strip()


def _info_from_polycard(card: dict) -> dict:
    meta = card.get("metadata") or {}
    comps = _components(card)
    price = comps.get("price") or {}

    current = (price.get("current_price") or {}).get("value")
    previous = (price.get("previous_price") or {}).get("value")
    label = (price.get("discount_label") or {}).get("text") or ""
    is_pix = "pix" in label.lower()

    # Analisa o bloco 'installments' (pode ser "ou R$ X em outros meios" OU "em 10x R$ X sem juros")
    inst = price.get("installments") or {}
    inst_raw_text = inst.get("text") or ""
    inst_values = inst.get("values") or []
    inst_no_interest = inst.get("no_interest")  # True = sem juros, False = com juros

    other_means_price = None
    installments_text = None

    if inst_raw_text:
        formatted_inst = _format_interpolated_text(inst_raw_text, inst_values)
        if "outros meios" in inst_raw_text.lower():
            # É o preço normal fora do Pix ("ou R$ 329,99 em outros meios")
            for v in inst_values:
                if (
                    isinstance(v, dict)
                    and (v.get("price") or {}).get("value") is not None
                ):
                    other_means_price = float(v["price"]["value"])
                    break
        else:
            # É informação de parcelamento real (ex: "em 10x R$ 32,99 sem juros")
            installments_text = formatted_inst
            if (
                inst_no_interest is True
                and "sem juros" not in installments_text.lower()
            ):
                installments_text += " sem juros"
            elif inst_no_interest is False and "juros" not in installments_text.lower():
                installments_text += " com juros"

    pics = (card.get("pictures") or {}).get("pictures") or []
    pic_id = pics[0].get("id") if pics and isinstance(pics[0], dict) else None

    url = meta.get("url")
    product_url = None
    if url:
        clean_u = url.replace("\\u002F", "/").lstrip("/")
        if not clean_u.startswith("http"):
            clean_u = "https://" + clean_u
        product_url = (
            clean_u + (meta.get("url_params") or "") + (meta.get("url_fragments") or "")
        )

    seller = (comps.get("seller") or {}).get("text")
    if seller:
        seller = re.sub(r"\s*\{[^}]*\}", "", seller).strip()

    regular_price = (
        other_means_price if (is_pix and other_means_price is not None) else current
    )
    pix_price = current if is_pix else None

    return {
        "itemId": meta.get("id")
        or meta.get("user_product_id")
        or meta.get("product_id"),
        "title": (comps.get("title") or {}).get("text"),
        "price": regular_price,
        "pix_price": pix_price,
        "original_price": previous,
        "discount_label": label or None,
        "installments_text": installments_text,
        "installments_no_interest": inst_no_interest,
        "thumbnail": (
            f"https://http2.mlstatic.com/D_NQ_NP_{pic_id}-O.jpg" if pic_id else None
        ),
        "seller": seller,
        "product_url": product_url,
    }


def _parse_direct_product_html(html: str, final_url: str = "") -> dict | None:
    """Fallback para quando a página aberta já é a página direta do produto (PDP)."""
    title = None
    image_url = None
    price = None
    pix_price = None
    original_price = None
    discount_label = None
    installments_text = None
    installments_no_interest = None
    item_id = None

    # 1. Tenta JSON-LD oficial presente na página do produto
    ld_matches = re.findall(
        r'<script[^>]*type="application/ld\+json"[^>]*>(.*?)</script>',
        html,
        flags=re.DOTALL,
    )
    for ld_raw in ld_matches:
        try:
            ld_data = json.loads(ld_raw.strip())
            if isinstance(ld_data, dict) and ld_data.get("@type") == "Product":
                title = ld_data.get("name")
                img = ld_data.get("image")
                image_url = img[0] if isinstance(img, list) and img else img
                offers = ld_data.get("offers") or {}
                if isinstance(offers, dict) and offers.get("price") is not None:
                    price = float(offers["price"])
                item_id = ld_data.get("sku") or ld_data.get("productID")
                break
        except Exception:
            pass

    # 2. Meta tags OpenGraph
    if not title:
        m_t = re.search(r'<meta\s+property="og:title"\s+content="([^"]+)"', html)
        if m_t:
            title = html_lib.unescape(m_t.group(1)).split(" - R$")[0].strip()

    if not image_url:
        m_i = re.search(r'<meta\s+property="og:image"\s+content="([^"]+)"', html)
        if m_i:
            image_url = m_i.group(1)

    # 3. Inspeciona o _n.ctx.r da página de produto (PDP) para pegar original_price, Pix e parcelas
    ctx = _extract_ctx(html)
    if ctx:
        for node in _walk(ctx):
            # Preço original riscado na PDP
            if original_price is None and isinstance(
                node.get("original_price"), (int, float)
            ):
                original_price = float(node["original_price"])
            if (
                price is None
                and isinstance(node.get("price"), (int, float))
                and node.get("currency_id") == "BRL"
            ):
                price = float(node["price"])

    if not title and price is None:
        return None

    return {
        "itemId": item_id,
        "title": title,
        "price": price,
        "pix_price": pix_price,
        "original_price": original_price,
        "discount_label": discount_label,
        "installments_text": installments_text,
        "installments_no_interest": installments_no_interest,
        "thumbnail": image_url,
        "seller": None,
        "product_url": final_url or None,
    }


def parse_social_html(html: str, final_url: str = "") -> dict | None:
    ctx = _extract_ctx(html)
    data = (((ctx or {}).get("appProps") or {}).get("pageProps") or {}).get(
        "data"
    ) or {}

    shared_ids = [
        s.get("id")
        for s in (data.get("shared_items") or [])
        if isinstance(s, dict) and s.get("id")
    ]
    if not shared_ids:
        m = re.search(r'"shared_item"\s*:\s*\{\s*"id"\s*:\s*"(MLBU?\d+)"', html)
        if m:
            shared_ids = [m.group(1)]

    card = _find_featured_polycard(data, shared_ids)

    if card:
        info = _info_from_polycard(card)
        info["title"] = info.get("title") or _meta_tag(data, "title")
        info["thumbnail"] = info.get("thumbnail") or _meta_tag(data, "image")
        if not info.get("product_url") and shared_ids:
            sid = shared_ids[0]
            if sid.startswith("MLB") and not sid.startswith("MLBU"):
                info["product_url"] = (
                    f"https://produto.mercadolivre.com.br/MLB-{sid[3:]}"
                )
        return info

    # Se não for uma página /social/ com polycard, tenta ler como página direta do produto (PDP)
    return _parse_direct_product_html(html, final_url=final_url)


def fetch_social_product(url: str, debug: bool = False) -> dict:
    """Baixa a página (/social/... ou PDP) e devolve os dados estruturados do produto."""
    cookie = os.environ.get("ML_AFFILIATE_COOKIE")
    headers = {
        "user-agent": USER_AGENT,
        "accept": "text/html,application/xhtml+xml",
        "accept-language": "pt-BR,pt;q=0.9",
    }
    if cookie:
        # Remove aspas simples extras caso estejam salvas no .env
        headers["cookie"] = cookie.strip().strip("'")

    r = requests.get(url, headers=headers, timeout=20, allow_redirects=True)
    if debug:
        print(f"   GET status={r.status_code} final={r.url[:110]} bytes={len(r.text)}")

    if "account-verification" in r.url or "webdevice/config" in r.url:
        raise SocialPageBlocked(
            "O ML pediu verificação em vez de mostrar a página "
            f"(redirecionou para: {r.url[:100]})."
        )

    info = parse_social_html(r.text, final_url=r.url)
    if not info:
        raise ValueError(
            "Não achei os dados do produto nessa página (pode ser um perfil sem produto "
            "ou uma página de verificação do ML)."
        )
    return info

def extract_title_from_formatted_message(formatted_msg: str) -> str:
    """Extrai o título limpo do produto a partir da 1ª linha da mensagem formatada."""
    if not formatted_msg:
        return "Sem título"
    first_line = formatted_msg.strip().splitlines()[0]
    no_tags = re.sub(r"<[^>]+>", "", first_line)
    cleaned = re.sub(r"^[🔥🏷️🎟️⚡💥🛒✨\s]+", "", no_tags).strip()
    return cleaned or "Produto sem título"


def normalize_and_tokenize(title: str) -> tuple[str, list[str], set[str]]:
    """Normaliza o título e retorna (string_limpa, lista_ordenada, set_de_tokens)."""
    if not title:
        return "", [], set()

    nfkd = unicodedata.normalize("NFKD", title)
    no_accent = "".join(c for c in nfkd if not unicodedata.combining(c)).lower()

    clean_str = re.sub(r"[^a-z0-9\s]", " ", no_accent)
    raw_tokens = clean_str.split()

    unique_ordered = []
    seen = set()
    for t in raw_tokens:
        if len(t) <= 2 and not any(ch.isdigit() for ch in t):
            continue
        if t in STOPWORDS_PT:
            continue
        if t.isdigit() and int(t) < 50:
            continue
        if len(t) > 4 and t.endswith("s") and not t.endswith("ss"):
            t = t[:-1]
        if t not in seen:
            seen.add(t)
            unique_ordered.append(t)

    norm_joined = " ".join(unique_ordered)
    return norm_joined, unique_ordered, seen


def _build_token_profile(title: str) -> dict:
    """Cria um perfil pré-calculado do título para comparações em lote ultra-rápidas."""
    norm_str, token_list, token_set = normalize_and_tokenize(title)
    return {
        "title": title,
        "norm": norm_str,
        "list": token_list,
        "set": token_set,
        "head_set": set(token_list[:5]),
        "top3_list": token_list[:3],
        "top3_set": set(token_list[:3]),
        "head_sig": " ".join(token_list[:3]) if token_list else norm_str[:25],
    }


def _fast_similarity_from_profiles(prof_a: dict, prof_b: dict) -> float:
    """
    Calcula a similaridade usando os perfis já tokenizados em memória.
    Se não houver palavras em comum, retorna 0.0 instantaneamente sem rodar SequenceMatcher!
    """
    set_a = prof_a["set"]
    set_b = prof_b["set"]
    if not set_a or not set_b:
        return 0.0

    inter_len = len(set_a & set_b)
    if inter_len == 0:
        return 0.0

    # Atalho imediato: 3 primeiras palavras nucleares idênticas
    if len(prof_a["top3_list"]) >= 3 and len(prof_b["top3_list"]) >= 3:
        if prof_a["top3_list"] == prof_b["top3_list"]:
            return 88.0
        if prof_a["top3_set"] == prof_b["top3_set"]:
            return 82.0

    union_len = len(set_a | set_b)
    jaccard = inter_len / union_len if union_len else 0.0

    min_len = min(len(set_a), len(set_b))
    overlap = inter_len / min_len if min_len else 0.0

    head_a = prof_a["head_set"]
    head_b = prof_b["head_set"]
    head_inter = len(head_a & head_b)
    head_union = len(head_a | head_b)
    head_score = head_inter / head_union if head_union else 0.0

    # Filtro rápido: se mesmo sem SequenceMatcher a nota base for muito baixa, nem gasta CPU com SequenceMatcher
    partial_score = (head_score * 0.40) + (overlap * 0.30) + (jaccard * 0.15)
    if partial_score < 0.22:
        return round(partial_score * 100.0, 1)

    seq_ratio = SequenceMatcher(None, prof_a["norm"], prof_b["norm"]).ratio()
    final_score = (partial_score + (seq_ratio * 0.15)) * 100.0
    return round(min(100.0, final_score), 1)


def calculate_product_similarity(title_a: str, title_b: str) -> float:
    """Função pública para comparar dois títulos avulsos."""
    return _fast_similarity_from_profiles(
        _build_token_profile(title_a), _build_token_profile(title_b)
    )


def load_recent_queue_profiles(days: int = 7) -> list[dict]:
    """
    Carrega do SQLite todos os itens dos últimos N dias JÁ PRÉ-TOKENIZADOS em memória.
    Ideal para ser chamado 1 única vez antes do loop de geração de ofertas!
    """
    cutoff_date = (datetime.now(CUIABA_TZ) - timedelta(days=days)).strftime("%Y-%m-%d")
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute(
        "SELECT id, product_url, formatted_message, target_date FROM queue WHERE target_date >= ?",

    profiles = []
    for q_id, p_url, f_msg, t_date in rows:

        prof = _build_token_profile(title)

def is_product_recently_used_or_blocked(
    candidate_title: str,
    candidate_url: str = "",
    days: int | None = None,
    threshold: float | None = None,
    extra_accepted_titles: list[str] | None = None,
    preloaded_db_profiles: list[dict] | None = None,
) -> tuple[bool, str]:
    """
    Verifica rapidamente se um produto candidato é repetido ou bloqueado.
    Se `preloaded_db_profiles` for passado, faz 0 consultas ao SQLite dentro do loop!
    """
    if days is None:
        days = int(get_config("dedup_history_days") or "7")
    if threshold is None:
        threshold = float(get_config("dedup_similarity_threshold") or "60")

    blocked_csv = get_config("dedup_blocked_terms") or ""
    blocked_terms = [t.strip().lower() for t in blocked_csv.split(",") if t.strip()]

    cand_prof = _build_token_profile(candidate_title)
    raw_cand_lower = (candidate_title or "").lower()

    for b_term in blocked_terms:
        nfkd_b = unicodedata.normalize("NFKD", b_term)
        clean_b = (
            "".join(c for c in nfkd_b if not unicodedata.combining(c)).lower().strip()
        )
        if clean_b and (clean_b in raw_cand_lower or clean_b in cand_prof["norm"]):
            return True, f"Termo bloqueado: '{b_term}'"

    if extra_accepted_titles:
        for prev_title in extra_accepted_titles:
            sim_batch = _fast_similarity_from_profiles(
                cand_prof, _build_token_profile(prev_title)
            )
            if sim_batch >= threshold:
                return True, f"Duplicado no mesmo lote ({sim_batch}%)"

    db_profiles = (
        preloaded_db_profiles
        if preloaded_db_profiles is not None
        else load_recent_queue_profiles(days=days)
    )
    cand_url_clean = (candidate_url or "").strip()

    for existing_prof in db_profiles:
        if cand_url_clean and existing_prof["product_url"] == cand_url_clean:
            return True, f"Mesma URL já na fila #{existing_prof['id']}"

        sim = _fast_similarity_from_profiles(cand_prof, existing_prof)
        if sim >= threshold:
            return (
                True,
                f"Similar ({sim}%) ao item #{existing_prof['id']} ({existing_prof['target_date']})",
            )

    return False, ""


def analyze_repeated_offers(
    days: int = 14, threshold: float = 60.0, store_filter: str = "all"
) -> dict:
    """
    Analisa a tabela `queue` nos últimos `days` dias usando perfis pré-calculados em memória.
    Tempo médio para 400 itens: ~25 ms.
    """
    cutoff_date = (datetime.now(CUIABA_TZ) - timedelta(days=days)).strftime("%Y-%m-%d")

    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute(
        """
        SELECT id, store, product_url, formatted_message, media_url, status, priority, target_date, created_at
        FROM queue
        WHERE target_date >= ?
        ORDER BY target_date DESC, id DESC
        """,
        (cutoff_date,),
    )
    rows = cursor.fetchall()
    conn.close()

    items = []
    for r in rows:
        q_id, store, prod_url, f_msg, media_url, status, prio, t_date, c_at = r
        if store_filter != "all" and store_filter.lower() not in (store or "").lower():
            continue

        title = extract_title_from_formatted_message(f_msg)
        prof = _build_token_profile(title)
        items.append(
            {
                "id": q_id,
                "store": store,
                "product_url": (prod_url or "").strip(),
                "title": title,
                "norm_title": prof["norm"],
                "head_signature": prof["head_sig"],
                "profile": prof,
                "media_url": media_url,
                "status": status,
                "priority": prio,
                "target_date": t_date,
                "created_at": c_at,
            }
        )

    clusters = []
    visited = set()
    n_items = len(items)

    for i in range(n_items):
        item_a = items[i]
        if item_a["id"] in visited:
            continue

        prof_a = item_a["profile"]
        url_a = item_a["product_url"]

        group = [
            {k: v for k, v in item_a.items() if k != "profile"} | {"similarity": 100.0}
        ]
        visited.add(item_a["id"])

        for j in range(i + 1, n_items):
            item_b = items[j]
            if item_b["id"] in visited:
                continue

            if url_a and url_a == item_b["product_url"]:
                sim = 100.0
            else:
                sim = _fast_similarity_from_profiles(prof_a, item_b["profile"])

            if sim >= threshold:
                group.append(
                    {k: v for k, v in item_b.items() if k != "profile"}
                    | {"similarity": sim}
                )
                visited.add(item_b["id"])

        if len(group) > 1:
            dates_set = sorted({g["target_date"] for g in group}, reverse=True)
            pending_ids = [g["id"] for g in group if g["status"] == "pending"]
            sent_count = sum(1 for g in group if g["status"] == "sent")

            if sent_count > 0:
                suggested_delete_ids = pending_ids
            else:
                suggested_delete_ids = pending_ids[:-1] if len(pending_ids) > 1 else []

            clusters.append(
                {
                    "canonical_title": item_a["title"],
                    "signature": item_a["head_signature"],
                    "media_url": item_a["media_url"],
                    "total_occurrences": len(group),
                    "sent_count": sent_count,
                    "pending_count": len(pending_ids),
                    "dates": dates_set,
                    "items": group,
                    "suggested_delete_ids": suggested_delete_ids,
                }
            )

    clusters.sort(
        key=lambda c: (c["total_occurrences"], c["pending_count"]), reverse=True
    )

    total_suggested_deletes = sum(len(c["suggested_delete_ids"]) for c in clusters)

    return {
        "total_scanned": n_items,
        "total_clusters": len(clusters),
        "total_duplicated_items": sum(c["total_occurrences"] for c in clusters),
        "total_suggested_deletes": total_suggested_deletes,
        "clusters": clusters,
    }

@app.post("/send-now/{offer_id}/{target}")
async def send_now(offer_id: int, target: str):
    conn = sqlite3.connect(DB_NAME)
    cursor = conn.cursor()
    cursor.execute(
        "SELECT store, product_url, formatted_message, media_url, COALESCE(media_type, 'image') FROM queue WHERE id = ?",
        (offer_id,),
    )
    row = cursor.fetchone()

    if row:
        store, product_url, saved_msg, saved_media, media_type = row
        if product_url:
            formatted_message, fresh_image = refresh_offer_data(product_url)
            media_url = (
                saved_media if media_type == "video" and saved_media else fresh_image
            )
            cursor.execute(
                "UPDATE queue SET formatted_message = ?, media_url = ? WHERE id = ?",
                (formatted_message, media_url, offer_id),
            )
            conn.commit()
        else:
            formatted_message, media_url = saved_msg, saved_media

        success = False
        if target == "all":
            s1 = trigger_whatsapp_dispatch(
                WHATSAPP_GROUP_PRINCIPAL, formatted_message, media_url, media_type
            )
            time.sleep(1)
            s2 = trigger_whatsapp_dispatch(
                WHATSAPP_GROUP_PARCERIA, formatted_message, media_url, media_type
            )
            success = s1 or s2
        elif target == "principal":
            success = trigger_whatsapp_dispatch(
                WHATSAPP_GROUP_PRINCIPAL, formatted_message, media_url, media_type
            )
        elif target == "parceria":
            success = trigger_whatsapp_dispatch(
                WHATSAPP_GROUP_PARCERIA, formatted_message, media_url, media_type
            )

        if success:
            cursor.execute("UPDATE queue SET status = 'sent' WHERE id = ?", (offer_id,))
            conn.commit()
    conn.close()
    return RedirectResponse(url="/", status_code=303)

        video_url_found = None
        video_status_param = ""

            video_url_found = manual_video_url.strip()

class ShopeeAffiliateAPI:
    GRAPHQL_URL = "https://open-api.affiliate.shopee.com.br/graphql"

    def __init__(self, app_id: str, app_secret: str):
        self.app_id = app_id
        self.app_secret = app_secret

    def _sign(self, payload: str, timestamp: str) -> str:
        factor = f"{self.app_id}{timestamp}{payload}{self.app_secret}"
        return hashlib.sha256(factor.encode("utf-8")).hexdigest()

    def _graphql_request(self, query: str) -> dict:
        payload_str = json.dumps({"query": query})
        timestamp = str(int(time.time()))
        signature = self._sign(payload_str, timestamp)

        headers = {
            "Authorization": f"SHA256 Credential={self.app_id}, Timestamp={timestamp}, Signature={signature}",
            "Content-Type": "application/json",
        }

        try:
            resp = requests.post(
                self.GRAPHQL_URL,
                data=payload_str.encode("utf-8"),
                headers=headers,
                timeout=20,
            )
            resp.raise_for_status()
            data = resp.json()
            if "errors" in data:
                logger.warning(f"Aviso GraphQL Shopee: {data['errors']}")
                return {}
            return data.get("data", {})
        except Exception as e:
            logger.error(f"Erro na requisição GraphQL Shopee: {e}")
            return {}

    def get_product_offer(self, item_id: int) -> dict | None:
        query = f"""query {{
            productOfferV2(itemId: {item_id}) {{
                nodes {{
                    itemId
                    productName
                    priceMin
                    priceMax
                    priceDiscountRate
                    ratingStar
                    sales
                    shopName
                    productLink
                    offerLink
                    imageUrl
                }}
            }}
        }}"""
        data = self._graphql_request(query)
        nodes = data.get("productOfferV2", {}).get("nodes", [])
        return nodes[0] if nodes else None

    def get_top_performing_offers(
        self, page: int = 1, limit: int = 50
    ) -> tuple[list[dict], bool]:
        """Busca ofertas de alta performance (listType = 2 = TOP_PERFORMING)."""
        query = f"""query {{
            productOfferV2(listType: 2, page: {page}, limit: {limit}) {{
                nodes {{
                    itemId
                    productName
                    priceMin
                    priceMax
                    priceDiscountRate
                    ratingStar
                    sales
                    shopName
                    productLink
                    offerLink
                    imageUrl
                }}
                pageInfo {{
                    page
                    limit
                    hasNextPage
                }}
            }}
        }}"""
        data = self._graphql_request(query)
        conn_data = data.get("productOfferV2", {})
        nodes = conn_data.get("nodes", [])
        has_next = bool(conn_data.get("pageInfo", {}).get("hasNextPage", False))
        return nodes, has_next

    async def extract_shopee_video_fast(self, browser, product_url: str) -> str | None:
        """Extrai o link .mp4 da Shopee com timeout rígido para nunca travar."""
        if not browser:
            return None

        captured_video_url = None
        context = None

        async def _run_extraction():
            nonlocal captured_video_url, context
            context = await browser.new_context(
                user_agent="Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36",
                viewport={"width": 1366, "height": 768},
            )
            page = await context.new_page()

            await page.add_init_script("""
                window.__captured_video = null;
                const checkVideo = () => {
                    const v = document.querySelector('video, video source');
                    if (v && (v.src || v.currentSrc)) {
                        window.__captured_video = v.src || v.currentSrc;
                    }
                };
                const observer = new MutationObserver(checkVideo);
                observer.observe(document.documentElement, { childList: true, subtree: true, attributes: true });
            """)

            def on_request(request):
                nonlocal captured_video_url
                req_url = request.url
                if ".mp4" in req_url and (
                    "susercontent.com" in req_url or "shopee" in req_url
                ):
                    captured_video_url = req_url.split("?")[0]

            async def on_response(response):
                nonlocal captured_video_url
                if captured_video_url:
                    return
                res_url = response.url
                if ".mp4" in res_url and (
                    "susercontent.com" in res_url or "shopee" in res_url
                ):
                    captured_video_url = res_url.split("?")[0]
                    return
                if (
                    "pdp/get" in res_url or "item/get" in res_url
                ) and response.status == 200:
                    try:
                        body = await asyncio.wait_for(response.text(), timeout=2.0)
                        match = re.search(r'https?://[^"\'\s\\]+\.mp4', body)
                        if match:
                            captured_video_url = (
                                match.group(0)
                                .replace("\\u002F", "/")
                                .replace("\\/", "/")
                            )
                    except Exception:
                        pass

            page.on("request", on_request)
            page.on("response", on_response)

            try:
                await page.goto(product_url, wait_until="commit", timeout=8000)
            except Exception:
                pass

            for _ in range(12):
                if captured_video_url:
                    break

                if "/buyer/login" in page.url or "/verify/" in page.url:
                    break

                try:
                    found = await page.evaluate(r"""() => {
                        if (window.__captured_video) return window.__captured_video;
                        const v = document.querySelector('video, video source');
                        if (v && (v.src || v.currentSrc)) return v.src || v.currentSrc;
                        const html = document.documentElement.innerHTML;
                        const m = html.match(/https?:\/\/[^"'\s\\]+\.mp4/);
                        return m ? m[0] : null;
                    }""")
                    if found and "http" in found:
                        captured_video_url = found.replace("\\u002F", "/").replace(
                            "\\/", "/"
                        )
                        break
                except Exception:
                    pass

                await asyncio.sleep(0.5)

            return captured_video_url

        try:
            return await asyncio.wait_for(_run_extraction(), timeout=10.0)
        except Exception:
            return captured_video_url
        finally:
            if context:
                try:
                    await context.close()
                except Exception:
                    pass

    def resolve_short_link(self, short_url: str) -> str:
        try:
            session = requests.Session()
            resp = session.get(
                short_url,
                headers={"User-Agent": "Mozilla/5.0"},

                timeout=10,
            )

    def extract_item_id(self, url: str) -> int | None:
        match_param = re.search(r"[?&](?:itemId|item_id)=(\d+)", url, re.IGNORECASE)
        if match_param:
            return int(match_param.group(1))
        match_path = re.search(r"i\.\d+\.(\d+)", url) or re.search(
            r"-i\.\d+\.(\d+)", url
        )
        if match_path:
            return int(match_path.group(1))
        match_product = re.search(r"/product/\d+/(\d+)", url) or re.search(
            r"/sp/product/(\d+)", url
        )
        if match_product:
            return int(match_product.group(1))
        match_end_id = re.search(r"/(\d+)(?:\?|$)", url)
        if match_end_id:
            return int(match_end_id.group(1))
        return None

            p_max = float(price_max) if price_max else p_min

                msg += (

                    f"💰 <b>Preço:</b> R$ {p_min:,.2f}\n\n".replace(",", "X")

                    .replace("X", ".")

            offer, store_name=store_name, extra_coupon_links=extra_coupon_links
        )
        msg += f"\n👉 {link}"
        return msg

    def get_category_offers_pool(
        self, categories_csv: str, max_pages_per_term: int = 3
    ) -> list[dict]:
        """
        Método auxiliar de compatibilidade (usado também pelo script de diagnóstico).
        """
        raw_categories = [
            c.strip() for c in (categories_csv or "").split(",") if c.strip()
        ]
        if not raw_categories:
            raw_categories = [
                "Roupas Femininas",
                "Moda Infantil",
                "Sapatos Femininos",
            ]

        pool: list[dict] = []
        for cat in raw_categories:
            cat_lower = cat.lower()
            subterms = CATEGORY_SEARCH_EXPANSION.get(cat_lower, [cat])
            chosen_terms = random.sample(subterms, k=min(2, len(subterms)))

            for term in chosen_terms:
                page = 1
                pages_fetched = 0
                while pages_fetched < max_pages_per_term:
                    nodes, has_next = self.get_best_selling_offers(
                        keyword=term, page=page, limit=50
                    )
                    if not nodes:
                        break
                    for n in nodes:
                        n["_source_bucket"] = cat
                    pool.extend(nodes)
                    pages_fetched += 1
                    if not has_next:
                        break
                    page += 1

        random.shuffle(pool)
        return pool

    def _harvest_valid_offers_deep(
        self,
        target_date: str,
        dedup_days: int,
        dedup_threshold: float,
        general_target: int,
        categories_target: int,
        categories_csv: str,
        keywords_csv: str,
    ) -> tuple[list[dict], list[dict], list[dict]]:
        """
        Varre páginas profundas da API da Shopee (usando hasNextPage real, sem trava em 5 ou 10 páginas)
        filtrando contra o banco de dados até acumular um amplo pool de ofertas INÉDITAS para sortear.
        Retorna: (valid_general_pool, valid_categories_pool, all_raw_backup_nodes)
        """
        preloaded_profiles = load_recent_queue_profiles(days=dedup_days)
        seen_ids: set[int] = set()
        all_raw_backup: list[dict] = []

        # Queremos acumular pelo menos 4x a quantidade necessária de ofertas inéditas
        # para que o sorteio final seja realmente aleatório entre dezenas de opções válidas!
        desired_general_pool_size = max(general_target * 4, 15)
        desired_cat_pool_size = max(categories_target * 4, 30)

        def _filter_unseen_nodes(
            nodes: list[dict], bucket_label: str, existing_pool_titles: list[str]
        ) -> list[dict]:
            approved = []
            for offer in nodes:
                item_id = offer.get("itemId")
                if not item_id or item_id in seen_ids:
                    continue
                seen_ids.add(item_id)

                product_name = (offer.get("productName") or "").strip()
                if not product_name:
                    continue

                offer["_source_bucket"] = bucket_label
                all_raw_backup.append(offer)

                product_url = (
                    offer.get("productLink")
                    or offer.get("offerLink")
                    or f"https://shopee.com.br/product/0/{item_id}"
                )

                if is_offer_recent(
                    product_url,
                    item_id=item_id,
                    days=dedup_days,
                    target_date=target_date,
                ):
                    continue

                is_dup, _ = is_product_recently_used_or_blocked(
                    candidate_title=product_name,
                    candidate_url=product_url,
                    days=dedup_days,
                    threshold=dedup_threshold,
                    extra_accepted_titles=existing_pool_titles,
                    preloaded_db_profiles=preloaded_profiles,
                )
                if is_dup:
                    continue

                existing_pool_titles.append(product_name)
                approved.append(offer)
            return approved

        # ------------------------------------------------------------------
        # 1. VARREDURA PROFUNDA DE CATEGORIAS (70%: Roupas Femininas, Moda Infantil, Sapatos Femininos)
        # ------------------------------------------------------------------
        raw_categories = [
            c.strip() for c in (categories_csv or "").split(",") if c.strip()
        ]
        if not raw_categories:
            raw_categories = [
                "Roupas Femininas",
                "Moda Infantil",
                "Sapatos Femininos",
            ]

        per_cat_desired = max(
            10, int(desired_cat_pool_size / max(1, len(raw_categories))) + 4
        )
        valid_by_cat: dict[str, list[dict]] = {cat: [] for cat in raw_categories}
        cat_accepted_titles: list[str] = []

        for cat in raw_categories:
            cat_lower = cat.lower()
            subterms = list(CATEGORY_SEARCH_EXPANSION.get(cat_lower, [cat]))
            random.shuffle(subterms)

            # Percorre os subtermos e avança nas páginas (1, 2, 3... até 35) enquanto precisar de inéditos
            for term in subterms:
                if len(valid_by_cat[cat]) >= per_cat_desired:
                    break

                page = 1
                while page <= 35 and len(valid_by_cat[cat]) < per_cat_desired:
                    nodes, has_next = self.get_best_selling_offers(
                        keyword=term, page=page, limit=50
                    )
                    if not nodes:
                        break

                    random.shuffle(nodes)
                    new_valid = _filter_unseen_nodes(nodes, cat, cat_accepted_titles)
                    valid_by_cat[cat].extend(new_valid)

                    if not has_next:
                        break
                    page += 1

            random.shuffle(valid_by_cat[cat])

        # Combina os candidatos inéditos das 3 categorias em formato Round-Robin (para balancear as 3 categorias)
        valid_categories_pool: list[dict] = []
        cat_keys = list(valid_by_cat.keys())
        while any(valid_by_cat[k] for k in cat_keys):
            random.shuffle(cat_keys)
            for k in cat_keys:
                if valid_by_cat[k]:
                    valid_categories_pool.append(valid_by_cat[k].pop(0))

        # ------------------------------------------------------------------
        # 2. VARREDURA PROFUNDA DO MODO GERAL (30%: TOP_PERFORMING + MAIS VENDIDOS)
        # ------------------------------------------------------------------
        valid_general_pool: list[dict] = []
        gen_accepted_titles: list[str] = list(cat_accepted_titles)
        keywords = [k.strip() for k in (keywords_csv or "").split(",") if k.strip()]

        # 2.A: Avança nas páginas de TOP_PERFORMING enquanto precisar de itens inéditos
        page_top = 1
        while page_top <= 35 and len(valid_general_pool) < (
            desired_general_pool_size // 2 + 5
        ):
            nodes, has_next = self.get_top_performing_offers(page=page_top, limit=50)
            if not nodes:
                break
            random.shuffle(nodes)
            valid_general_pool.extend(
                _filter_unseen_nodes(nodes, "Mais Vendidos / Top", gen_accepted_titles)
            )
            if not has_next:
                break
            page_top += 1

        # 2.B: Avança nas páginas de MAIS VENDIDOS (por palavra-chave ou geral)
        if keywords:
            random.shuffle(keywords)
            for kw in keywords:
                if len(valid_general_pool) >= desired_general_pool_size:
                    break
                page_kw = 1
                while (
                    page_kw <= 35
                    and len(valid_general_pool) < desired_general_pool_size
                ):
                    nodes, has_next = self.get_best_selling_offers(
                        keyword=kw, page=page_kw, limit=50
                    )
                    if not nodes:
                        break
                    random.shuffle(nodes)
                    valid_general_pool.extend(
                        _filter_unseen_nodes(
                            nodes, f"Busca: {kw.title()}", gen_accepted_titles
                        )
                    )
                    if not has_next:
                        break
                    page_kw += 1
        else:
            page_sold = 1
            while (
                page_sold <= 35 and len(valid_general_pool) < desired_general_pool_size
            ):
                nodes, has_next = self.get_best_selling_offers(
                    keyword=None, page=page_sold, limit=50
                )
                if not nodes:
                    break
                random.shuffle(nodes)
                valid_general_pool.extend(
                    _filter_unseen_nodes(
                        nodes, "Mais Vendidos / Top", gen_accepted_titles
                    )
                )
                if not has_next:
                    break
                page_sold += 1

        random.shuffle(valid_general_pool)
        return valid_general_pool, valid_categories_pool, all_raw_backup

    def generate_auto_offers_to_queue(
        self,
        count: int = 15,
        target_date: str | None = None,
        keywords_csv: str | None = None,
    ) -> int:
        """
        Gera ofertas automáticas da Shopee dividindo:
        - 30% (configurável em `shopee_general_pct`): Busca Geral (TOP_PERFORMING + Mais Vendidos)
        - 70% (restante): Categorias específicas (Roupas Femininas, Moda Infantil, Sapatos Femininos)
        Varre páginas profundas sem limite baixo e sorteia dentro dos inéditos.
        """
        if not target_date:
            target_date = datetime.now(cuiaba_tz).strftime("%Y-%m-%d")

        if keywords_csv is None:
            keywords_csv = get_config("auto_keywords") or ""

        general_pct = int(get_config("shopee_general_pct") or "30")
        general_pct = max(0, min(100, general_pct))
        target_categories_csv = (
            get_config("shopee_target_categories")
            or "Roupas Femininas, Moda Infantil, Sapatos Femininos"
        )

        dedup_days = int(get_config("dedup_history_days") or "7")
        dedup_threshold = float(get_config("dedup_similarity_threshold") or "60")

        general_target = round(count * (general_pct / 100.0))
        if general_pct > 0 and count >= 2 and general_target == 0:
            general_target = 1
        categories_target = max(0, count - general_target)

        logger.info(
            f"🔎 Iniciando varredura paginada profunda na Shopee: meta={count} ({general_target} Geral / {categories_target} Categorias)..."
        )

        valid_general_pool, valid_categories_pool, all_raw_backup = (
            self._harvest_valid_offers_deep(
                target_date=target_date,
                dedup_days=dedup_days,
                dedup_threshold=dedup_threshold,
                general_target=general_target,
                categories_target=categories_target,
                categories_csv=target_categories_csv,
                keywords_csv=keywords_csv,
            )
        )

        logger.info(
            f"🎯 Pools de inéditos prontos para sorteio: {len(valid_categories_pool)} de Categorias | {len(valid_general_pool)} Gerais."
        )

        # Sorteia dentro dos pools de inéditos
        chosen_categories = valid_categories_pool[:categories_target]
        remaining_cat = valid_categories_pool[categories_target:]

        chosen_general = valid_general_pool[:general_target]
        remaining_gen = valid_general_pool[general_target:]

        # Se uma das duas listas (30% ou 70%) teve menos inéditos que a cota, completa com a sobra da outra
        total_so_far = len(chosen_general) + len(chosen_categories)
        if total_so_far < count:
            missing = count - total_so_far
            leftovers = remaining_cat + remaining_gen
            chosen_categories.extend(leftovers[:missing])

        # VÁLVULA DE SEGURANÇA ANTI-TRAVAMENTO:
        # Se mesmo varrendo dezenas de páginas ainda faltarem itens para bater `count`,
        # relaxa a janela de repetição apenas para os itens faltantes (ex: não repetir no mesmo dia/últimos 2 dias)
        total_so_far = len(chosen_general) + len(chosen_categories)
        if total_so_far < count and all_raw_backup:
            missing = count - total_so_far
            logger.warning(
                f"⚠️ Pool estrito de {dedup_days}d rendeu {total_so_far}/{count} itens. Acionando fallback seguro para completar +{missing} ofertas!"
            )
            already_chosen_ids = {
                o.get("itemId") for o in (chosen_general + chosen_categories)
            }
            random.shuffle(all_raw_backup)
            for backup_offer in all_raw_backup:
                if len(chosen_general) + len(chosen_categories) >= count:
                    break
                b_id = backup_offer.get("itemId")
                if not b_id or b_id in already_chosen_ids:
                    continue
                b_url = (
                    backup_offer.get("productLink")
                    or backup_offer.get("offerLink")
                    or f"https://shopee.com.br/product/0/{b_id}"
                )
                # Garante que pelo menos não esteja na fila de hoje/ontem
                if is_offer_recent(
                    b_url, item_id=b_id, days=1, target_date=target_date
                ):
                    continue
                already_chosen_ids.add(b_id)
                chosen_categories.append(backup_offer)

        # Intercala as ofertas (2 de Categorias para 1 Geral) para distribuir ao longo do dia
        final_offers: list[dict] = []
        while chosen_categories or chosen_general:
            for _ in range(2):
                if chosen_categories:
                    final_offers.append(chosen_categories.pop(0))
            if chosen_general:
                final_offers.append(chosen_general.pop(0))

        added = 0
        for offer in final_offers:
            item_id = offer.get("itemId")
            product_url = (
                offer.get("productLink")
                or offer.get("offerLink")
                or f"https://shopee.com.br/product/0/{item_id}"
            )
            aff_link = (
                offer.get("offerLink")
                or self.generate_short_link(product_url)
                or product_url
            )

            msg = self.format_product_message(offer, aff_link)
            media_url = offer.get("imageUrl")
            category_name = offer.get("_source_bucket", "Mais Vendidos / Top")

            add_to_queue(
                store="Shopee",
                product_url=product_url,
                message=msg,
                media_url=media_url,
                media_type="image",
                priority=0,
                target_date=target_date,
                category=category_name,
            )
            added += 1

        logger.info(
            f"✅ Shopee Auto concluído: {added}/{count} ofertas inéditas sorteadas e inseridas na fila ({target_date})."
        )
        return added

    # ==========================================================================
    # RELATÓRIO DE VENDAS E COMISSÕES (CONVERSION REPORT)
    # ==========================================================================
    def get_sales_report(self, days: int = 30) -> dict:
        """
        Busca os pedidos e comissões gerados na Shopee via GraphQL `conversionReport`.
        """
        now_dt = datetime.now(cuiaba_tz)
        start_dt = now_dt - timedelta(days=days)
        purchase_time_start = int(start_dt.timestamp())
        purchase_time_end = int(now_dt.timestamp())

        query_full = f"""query {{
            conversionReport(purchaseTimeStart: {purchase_time_start}, purchaseTimeEnd: {purchase_time_end}, limit: 100) {{
                nodes {{
                    conversionId
                    purchaseTime
                    totalCommission
                    netCommission
                    buyerType
                    orders {{
                        orderId
                        orderStatus
                        items {{
                            itemId
                            itemName
                            itemPrice
                            actualAmount
                            qty
                            itemTotalCommission
                            imageUrl
                        }}
                    }}
                }}
                pageInfo {{
                    hasNextPage
                }}
            }}
        }}"""

        query_minimal = f"""query {{
            conversionReport(purchaseTimeStart: {purchase_time_start}, purchaseTimeEnd: {purchase_time_end}, limit: 100) {{
                nodes {{
                    conversionId
                    purchaseTime
                    totalCommission
                    orders {{
                        orderId
                        orderStatus
                        items {{
                            itemName
                            actualAmount
                            qty
                        }}
                    }}
                }}
            }}
        }}"""

        data = self._graphql_request(query_full)
        if not data or "conversionReport" not in data:
            data = self._graphql_request(query_minimal)

        nodes = (data.get("conversionReport") or {}).get("nodes") or []

        orders_list = []
        total_orders = 0
        pending_orders = 0
        completed_orders = 0
        cancelled_orders = 0
        total_gmv = 0.0
        total_commission_valid = 0.0
        completed_commission = 0.0
        pending_commission = 0.0

        def _to_float(val) -> float:
            if val is None or val == "":
                return 0.0
            try:
                return float(str(val).replace(",", "."))
            except Exception:
                return 0.0

        for node in nodes:
            p_time = node.get("purchaseTime")
            if p_time:
                try:
                    dt_str = datetime.fromtimestamp(int(p_time), tz=cuiaba_tz).strftime(
                        "%d/%m/%Y %H:%M"
                    )
                except Exception:
                    dt_str = str(p_time)
            else:
                dt_str = "-"

            conv_comm = _to_float(
                node.get("netCommission") or node.get("totalCommission")
            )
            node_orders = node.get("orders") or []

            if not node_orders:
                total_orders += 1
                pending_orders += 1
                total_commission_valid += conv_comm
                pending_commission += conv_comm
                orders_list.append(
                    {
                        "platform": "Shopee",
                        "order_id": str(node.get("conversionId") or "N/A"),
                        "purchase_time": dt_str,
                        "raw_ts": int(p_time or 0),
                        "status": "PENDING",
                        "status_label": "Pendente",
                        "items_summary": "Pedido registrado na conversão",
                        "items_count": 1,
                        "order_amount": 0.0,
                        "commission": conv_comm,
                        "buyer_type": node.get("buyerType") or "-",
                    }
                )
                continue

            for ord_obj in node_orders:
                order_id = ord_obj.get("orderId") or node.get("conversionId") or "N/A"
                raw_status = (ord_obj.get("orderStatus") or "PENDING").upper()
                items = ord_obj.get("items") or []

                order_gmv = 0.0
                items_comm = 0.0
                item_names = []
                total_qty = 0

                for it in items:
                    qty = int(it.get("qty") or 1)
                    total_qty += qty
                    actual_amt = _to_float(
                        it.get("actualAmount") or it.get("itemPrice")
                    )
                    order_gmv += actual_amt
                    items_comm += _to_float(it.get("itemTotalCommission"))
                    nm = (it.get("itemName") or "Produto Shopee").strip()
                    item_names.append(f"{qty}x {nm}")

                order_comm = (
                    items_comm
                    if items_comm > 0
                    else (conv_comm / max(1, len(node_orders)))
                )

                is_cancelled = any(
                    k in raw_status
                    for k in ("CANCEL", "INVALID", "UNPAID", "RETURN", "REJECT")
                )
                is_completed = any(
                    k in raw_status
                    for k in ("COMPLETE", "COMPLETED", "APPROVED", "PAID")
                )

                total_orders += 1
                if is_cancelled:
                    cancelled_orders += 1
                    status_label = "Cancelado"
                elif is_completed:
                    completed_orders += 1
                    status_label = "Concluído"
                    total_gmv += order_gmv
                    total_commission_valid += order_comm
                    completed_commission += order_comm
                else:
                    pending_orders += 1
                    status_label = "Pendente"
                    total_gmv += order_gmv
                    total_commission_valid += order_comm
                    pending_commission += order_comm

                orders_list.append(
                    {
                        "platform": "Shopee",
                        "order_id": str(order_id),
                        "purchase_time": dt_str,
                        "raw_ts": int(p_time or 0),
                        "status": raw_status,
                        "status_label": status_label,
                        "items_summary": (
                            " | ".join(item_names) if item_names else "Item Shopee"
                        ),
                        "items_count": total_qty or 1,
                        "order_amount": round(order_gmv, 2),
                        "commission": round(order_comm, 2),
                        "buyer_type": node.get("buyerType") or "-",
                    }
                )

        orders_list.sort(key=lambda x: x["raw_ts"], reverse=True)

        return {
            "days": days,
            "total_orders": total_orders,
            "pending_orders": pending_orders,
            "completed_orders": completed_orders,
            "cancelled_orders": cancelled_orders,
            "total_gmv": round(total_gmv, 2),
            "total_commission": round(total_commission_valid, 2),
            "completed_commission": round(completed_commission, 2),
            "pending_commission": round(pending_commission, 2),
            "orders": orders_list,
        }

"""
Serviço centralizado de envio de mensagens e mídias via Evolution API (WhatsApp).
Utilizado pelo bot do Telegram, pelo Portal Web e pelo despachante automático.
"""

import os
import logging
import requests
from dotenv import load_dotenv

load_dotenv()

logger = logging.getLogger(__name__)

EVOLUTION_API_URL = os.environ.get("EVOLUTION_API_URL", "http://localhost:8080").rstrip(
    "/"
)
EVOLUTION_API_KEY = os.environ.get("EVOLUTION_API_KEY", "")
EVOLUTION_INSTANCE = os.environ.get("EVOLUTION_INSTANCE", "chip_01")


def convert_html_to_whatsapp(text: str) -> str:
    if not text:
        return ""
    import re

    text = re.sub(r"<(?:b|strong)>(.*?)</(?:b|strong)>", r"*\1*", text, flags=re.DOTALL)
    text = re.sub(r"<(?:i|em)>(.*?)</(?:i|em)>", r"_\1_", text, flags=re.DOTALL)
    text = re.sub(r"<[^>]+>", "", text)
    return text


def send_whatsapp_message(
    group_id: str,
    message: str,
    media_url: str | None = None,
    media_type: str = "image",
) -> bool:
    """Envia texto ou mídia (imagem/vídeo) para um grupo do WhatsApp via Evolution API."""
    if not EVOLUTION_API_URL or not EVOLUTION_API_KEY:
        logger.error("❌ Credenciais da Evolution API não configuradas no .env.")
        return False

    final_message = convert_html_to_whatsapp(message)
    is_video = (media_type == "video") or (media_url and ".mp4" in media_url.lower())
    headers = {"apikey": EVOLUTION_API_KEY}

    try:
        if media_url:
            try:
                dl_headers = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64)"}
                media_resp = requests.get(media_url, headers=dl_headers, timeout=30)
                if media_resp.status_code == 200:
                    url_api = (
                        f"{EVOLUTION_API_URL}/message/sendMedia/{EVOLUTION_INSTANCE}"
                    )
                    if is_video:
                        files = {
                            "file": (
                                "oferta_video.mp4",
                                media_resp.content,
                                "video/mp4",
                            )
                        }
                        data = {
                            "number": group_id,
                            "mediatype": "video",
                            "caption": final_message,
                            "fileName": "oferta_video.mp4",
                        }
                    else:
                        files = {
                            "file": (
                                "oferta_qualidade.jpg",
                                media_resp.content,
                                "image/jpeg",
                            )
                        }
                        data = {
                            "number": group_id,
                            "mediatype": "image",
                            "caption": final_message,
                            "fileName": "oferta_qualidade.jpg",
                        }

                    res = requests.post(
                        url_api, data=data, files=files, headers=headers, timeout=60
                    )
                    if res.status_code in [200, 201]:
                        logger.info(
                            f"✅ Mídia enviada com sucesso para o WhatsApp ({group_id})"
                        )
                        return True
                    else:
                        logger.warning(
                            f"⚠️ Falha no envio de mídia ({res.text}), tentando fallback para texto..."
                        )
            except Exception as e:
                logger.warning(f"⚠️ Erro ao baixar mídia para envio: {e}")

        url_api = f"{EVOLUTION_API_URL}/message/sendText/{EVOLUTION_INSTANCE}"
        payload = {"number": group_id, "text": final_message, "delay": 1200}
        headers["Content-Type"] = "application/json"
        response = requests.post(url_api, json=payload, headers=headers, timeout=15)

        if response.status_code in [200, 201]:
            logger.info(f"✅ Texto enviado com sucesso para o WhatsApp ({group_id})")
            return True
        else:
            logger.error(f"❌ Erro ao enviar texto para o WhatsApp: {response.text}")
            return False
    except Exception as e:
        logger.error(f"❌ Exceção ao enviar mensagem para o WhatsApp: {e}")
        return False
